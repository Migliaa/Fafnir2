//-------------------------------------------------------------------------
// 165 DAX 30m  |  MODULO M5
// Operare solo quando la volatilità recente è sopra quella di lungo periodo
//
// Ipotesi. Il momento intraday, cioè la tendenza del pomeriggio a proseguire il
// movimento della mattina, è più forte quando la volatilità è alta: lo trovano Gao et
// al. 2018 sullo S&P 500 e Li-Sakkas-Urquhart 2022 su 16 borse sviluppate. Il
// meccanismo delle coperture gamma porta nella stessa direzione: i market maker di
// opzioni tendono ad avere esposizione gamma negativa dopo i ribassi e nei periodi
// agitati, e solo allora le loro coperture spingono il prezzo nella direzione del
// movimento (Baltussen et al. 2021). Nei periodi calmi gli short del pomeriggio vanno
// invece contro la salita di fondo dell'indice e contro gli acquisti sui ribassi.
// Qui un trade parte solo se la volatilità delle ultime 2 settimane è almeno SogliaM5
// volte quella degli ultimi 5 mesi circa.
// Direzione attesa: per lo short, guadagno medio dei trade tenuti chiaramente più alto
// di quello dei trade scartati. Per il long l'attesa è più debole, perché nei periodi
// agitati anche la notte è più rischiosa.
// È l'opposto di M2, che nei periodi agitati riduce la taglia: i due test insieme
// dicono se il guadagno per trade cresce con la volatilità più o meno del rischio.
// UsaM5 = 0 riproduce l'originale con Reinvest = 0 (salvo i primi giorni del backtest,
// per le barre precaricate).
//
// Varianti: SogliaM5 1.0 (proposta), 0.8 (scarta solo i periodi molto calmi), 1.2.
// Con 1.0 ci si aspetta di tenere poco meno della metà dei trade.
// Come provarlo: Reinvest = 0 (già impostato), così la taglia dipende solo dal prezzo
// d'ingresso e i trade scartati sono la differenza esatta tra originale e modulo.
// Lanciare UsaM5 = 0 e UsaM5 = 1, report long e short separati.
// Come giudicarlo, per ciascun lato: guadagno medio dei trade scartati =
//   (profitto originale - profitto M5) / (trade originale - trade M5)
// M5 funziona su un lato se questo valore è vicino a zero o negativo mentre quello dei
// trade tenuti sale. Togliendo metà dei trade il profitto totale scende comunque: conta
// il guadagno medio e il rapporto profitto / drawdown.
//
// Le righe GRAPH in fondo servono solo nel backtest: toglile prima del reale.
//-------------------------------------------------------------------------
defparam cumulateorders= false
DEFPARAM PreLoadBars = 5000

UsaM5 = 1                 // 0 = originale, 1 = modulo attivo
SogliaM5 = 1.0            // volatilità 2 settimane / volatilità 5 mesi
ApplicaLong  = 1
ApplicaShort = 1

// --- Volatilità giornaliera (barre giornaliere chiuse)
TIMEFRAME(daily, updateonclose)
VolBreve = AverageTrueRange[10](close)     // ultime 2 settimane
VolLunga = AverageTrueRange[100](close)    // ultimi 5 mesi circa
TIMEFRAME(default)

RegimeVol = 1
IF VolBreve > 0 AND VolLunga > 0 THEN
   RegimeVol = VolBreve / VolLunga
ENDIF

FiltroL = 1
FiltroS = 1
IF UsaM5 = 1 AND RegimeVol < SogliaM5 THEN
   IF ApplicaLong = 1 THEN
      FiltroL = 0
   ENDIF
   IF ApplicaShort = 1 THEN
      FiltroS = 0
   ENDIF
ENDIF

EntraLong=1
EntraShort=1

TagliaPosizione=1

MySLL=3
MyTPL=3.5
MySLS=2.5
MyTPS=1.4
//==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START - Strategy Param
SProfit             = Strategyprofit +900+550+150
Reinvest            = 0    // PER IL TEST (originale: 1)
InitCapital         = 6000  //Suggested Capital to start
RiskOnCapital       = 4 //Suggested %Factor to start buy 1 contract
RiskOnCapitalL      = 5
RiskOnProfit        = 12
MaxPosition         = 80    //Max contract (no more than 10 contract)

If Reinvest=0 then
SProfit=0
endif

PsL=((InitCapital)*(RiskOnCapitalL/100) + (SProfit)*(RiskOnProfit/100))/(close*(MySLL/100))
PsS=((InitCapital)*(RiskOnCapital/100) + (SProfit)*(RiskOnProfit/100))/(close*(MySLS/100))

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif

//==//==//==//==//==//==//==//==//==//==//==//==//==//==//==//

time2tradeL = time = 133000 and dayofweek <>5
time2tradeS = time = 123000 and dayofweek<>1
TriggerL = close > exponentialaverage [25] (close)
TriggerS = close < exponentialaverage [25] (close)

PatternL=rsi[2](Close)>65
PatternS=not (dlow(1)>dlow(5))

if EntraLong AND time2tradeL AND TriggerL AND PatternL AND FiltroL then
buy PsL Lots at market
endif

if EntraShort AND time2tradeS AND TriggerS AND PatternS AND FiltroS then
sellshort PsS Lots at market
endif

ES1=0

ExitL=(time=080000 ) or(dayofweek = 5 and time = 210000)
ExitS=ES1 OR Barindex-tradeindex>12

If LongOnMarket AND ExitL then
sell at market
endif

If ShortOnMarket AND ExitS then
exitshort at market
endif
If LongOnMarket AND MySLL>0 then
set stop %loss MySLL
endif

If ShortOnMarket AND MySLS>0 then
set stop %loss MySLS
endif

If LongOnMarket AND MyTPL>0 then
set target %profit MyTPL
endif

If ShortOnMarket AND MyTPS>0 then
set target %profit MyTPS
endif

// --- Solo backtest: controllo del modulo (togli prima del reale)
GRAPH RegimeVol AS "volatilità 2 settimane / 5 mesi"
