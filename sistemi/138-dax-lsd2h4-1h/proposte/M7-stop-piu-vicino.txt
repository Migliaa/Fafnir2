// 138 DAX LSD2H4 1H  |  MODULO M7, sopra C1 (originale + ingresso alle 09:00)
// Stop all'1,5% invece che al 3%, con la stessa posizione (variante: 2%)
//
// Ipotesi. Lo stop al 3% non è mai scattato: nel riferimento la perdita peggiore è
// stata l'1,05% della posizione, in C1 lo 0,99%. È uno stop di emergenza, ma è quello che
// decide il rischio scritto nella console: a 1 contratto, con il DAX a 26.700, uno stop
// vale circa 800 €, il 27% di 3.000. Uno stop più vicino ai movimenti reali di una
// giornata dimezza la perdita massima possibile in un giorno di forte rialzo (il caso che
// il backtest non ha ancora visto) e permette, a parità di rischio scritto, di scegliere
// la taglia con più margine.
// Il costo: uno stop taglia anche i trade che vanno contro e poi tornano (Kaminski-Lo
// 2014: con prezzi senza memoria uno stop riduce sempre il rendimento atteso). Il
// massimo movimento contrario dentro la giornata supera la perdita di chiusura, quindi
// qualche trade verrà fermato.
// Direzione attesa: all'1,5% pochi trade toccati, profitto quasi uguale (entro il 5-10%),
// drawdown uguale o più basso. Al 2% praticamente identico a C1. Se l'1,5% toglie molto,
// gli short del martedì vanno spesso contro prima di scendere, e lo stop resta lontano.
//
// La posizione resta da 30.000 € in tutte le varianti: il modulo riduce RischioShort in
// proporzione allo stop, così cambia solo l'uscita.
// Varianti (StopM7):
//   1.5   (proposta)
//   2     variante
// Riferimento: UsaM7 = 0, cioè C1 a posizione fissa = M2 della prima tornata
// (71 trade, 102,65 € a trade, 7.288,37 €, drawdown −810,97 €, peggior trade −297,83 €).
// Come giudicarlo: stessi ingressi; numero di perdenti a stop pieno (−450 € all'1,5%),
// guadagno medio, drawdown, profitto / drawdown (C1: 9,0).
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

//=========================================================================
// CONSOLE DI GESTIONE  -  tutto in euro (Germany 40 Cash da 1 €)
//
// 1. SOLDI: CapitaleVersato = soldi messi nel sistema; ProfittiPrecedenti = guadagni
//    delle esecuzioni precedenti, da riportare a mano a ogni riavvio.
//    Backtest: CapitaleVersato = capitale del backtest, ProfittiPrecedenti = 0.
CapitaleVersato     = 10000    // PER IL TEST (capitale del backtest 10.000 €)
ProfittiPrecedenti  = 0
// 2. RISCHIO PER TRADE: % del capitale persa se scatta lo stop
RischioShort        = 9        // PER IL TEST: 900 € con lo stop al 3% = posizione da 30.000 €
// 3. MANOPOLA DEL REINVESTIMENTO: 0 = rischio fisso in euro, 1 = normale, 2 = profitti doppi
Aggressivita        = 0        // PER IL TEST: stesso rischio in euro per ogni trade
// 4. LIMITI
MinContratti        = 1
MaxContratti        = 80
//=========================================================================

// --- Parametri M7
UsaM7          = 1             // 0 = C1, 1 = modulo attivo
StopM7         = 1.5           // stop dello short, % del prezzo (C1: 3)

// --- Parametri della strategia (C1)
EntraShort     = 1
Escludi2022    = 1             // 1 = come l'originale, per riprodurre M2
OraSegnaleS    = 040000
OraIngressoS   = 090000
OraUscitaS     = 210000

MySLS          = 3
MyTPS          = 3
SLbrkevenPS    = 2.5
TPbrkevenPS    = 2.5

// --- M7: stop più vicino, stessa posizione
IF UsaM7 = 1 THEN
   RischioShort = RischioShort * StopM7 / MySLS
   MySLS = StopM7
ENDIF

// --- Posizione (dalla console)
StopS = MySLS / 100
Profitti = ProfittiPrecedenti + StrategyProfit
IF Profitti > 0 THEN
   CapitaleBase = CapitaleVersato + Aggressivita * Profitti
ELSE
   CapitaleBase = CapitaleVersato + Profitti
ENDIF
RischioEuroS = max(0, CapitaleBase) * RischioShort / 100
PsS = max(MinContratti, min(MaxContratti, RischioEuroS / (close * StopS)))

GiornoOk = DayOfWeek = 2

// --- Segnale delle 04:00
Time2Trade = 1
IF Escludi2022 = 1 AND (month = 4 OR month = 7) AND year = 2022 THEN
   Time2Trade = 0
ENDIF

bptns08   = low < open[1] - 0.3 * (high[1] - low[1])
range1d   = DHigh(1) - DLow(1)
body1d    = abs(DOpen(1) - DClose(1))
uptn008   = body1d > 0.5 * range1d
uptn141   = (close[1] < open[1] AND close[2] > open[2])

ONCE SegnaleMemo = 0
IF Date <> Date[1] THEN
   SegnaleMemo = 0
ENDIF
IF GiornoOk AND time = OraSegnaleS THEN
   SegnaleMemo = bptns08 AND uptn141 AND NOT uptn008
ENDIF

// --- Ingresso alle 09:00
ONCE GiornoTrade = 0
IF Time2Trade AND EntraShort AND SegnaleMemo AND GiornoOk AND time = OraIngressoS AND NOT OnMarket THEN
   SELLSHORT PsS LOTS AT MARKET
   SET STOP %LOSS MySLS
   SET TARGET %PROFIT MyTPS
   GiornoTrade = Date
ENDIF

// --- Stop e target spostati all'ingresso
IF ShortOnMarket AND SLbrkevenPS > 0 THEN
   IF (tradeprice - low) / tradeprice * 100 >= SLbrkevenPS THEN
      SET STOP BREAKEVEN
   ENDIF
ENDIF
IF ShortOnMarket AND TPbrkevenPS > 0 THEN
   IF (high - tradeprice) / tradeprice * 100 >= TPbrkevenPS THEN
      SET TARGET BREAKEVEN
   ENDIF
ENDIF

// --- Uscita alle 21:00, o al primo cambio di giorno se manca quella barra
ETimeS = (barindex - tradeindex > 17) OR time = OraUscitaS
IF ShortOnMarket AND (ETimeS OR Date <> GiornoTrade) THEN
   EXITSHORT AT MARKET
ENDIF
