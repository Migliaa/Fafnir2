//-------------------------------------------------------------------------
// 138 DAX LSD2H4 1H  |  MODULO M6, sopra C1 (originale + ingresso alle 09:00)
// Lo short esce alle 17:00, prima della chiusura di Xetra (variante: alle 15:00)
//
// Ipotesi. Con C1 lo short sta in posizione dalle 09:00 alle 21:00 ed esce con lo
// spread serale di IG (3,8 punti invece di 1,6). Il tratto 17:00-21:00 non è stato
// misurato (la variante di M2 con uscita alle 17:00 non è stata lanciata). Il momento
// intraday dice che il movimento della giornata prosegue fino alla chiusura (Gao et al.
// 2018; Baltussen et al. 2021), e M3 ha mostrato che lo short rende soprattutto nei
// periodi agitati, come una gamba di momento: la sera dovrebbe rendere poco o niente, e
// le ore della seduta americana fino alle 17:00 (dati USA alle 14:30, apertura di Wall
// Street alle 15:30) dovrebbero contare.
// Direzione attesa: uscita alle 17:00 con guadagno medio quasi uguale e drawdown più
// basso, più 1,1 punti a contratto di spread risparmiato in reale. Uscita alle 15:00 con
// guadagno medio più basso, perché toglie l'apertura americana. Se invece le 17:00
// tolgono molto, il calo prosegue nella sera e si tiene fino alle 21:00.
//
// Varianti (OraUscitaM6):
//   170000   ultima mezz'ora di Xetra, spread 1,6 (proposta)
//   150000   prima dei dati e dell'apertura USA: misura quanto rende la seduta americana
// Nelle circa tre settimane l'anno con ora legale sfasata l'apertura USA è alle 14:30.
// Riferimento: UsaM6 = 0, cioè C1 a posizione fissa = M2 della prima tornata
// (71 trade, 102,65 € a trade, 7.288,37 €, drawdown −810,97 €).
// Come giudicarlo: stessi trade, quindi (profitto − 7.288,37) / 71 è quanto rende a
// trade il tratto tolto; poi drawdown e profitto / drawdown (C1: 9,0).
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

//=========================================================================
// CONSOLE DI GESTIONE  -  tutto in euro (Germany 40 Cash da 1 €)
//
// 1. SOLDI: CapitaleVersato = soldi messi nel sistema; ProfittiPrecedenti = guadagni
//    delle esecuzioni precedenti, da riportare a mano a ogni riavvio.
//    Backtest: CapitaleVersato = capitale del backtest, ProfittiPrecedenti = 0.
CapitaleVersato     = 10000    // PER IL TEST (capitale del backtest 10.000 €)
ProfittiPrecedenti  = 0
// 2. RISCHIO PER TRADE: % del capitale persa se scatta lo stop
RischioShort        = 9        // PER IL TEST: 900 € con lo stop al 3% = posizione da 30.000 €
// 3. MANOPOLA DEL REINVESTIMENTO: 0 = rischio fisso in euro, 1 = normale, 2 = profitti doppi
Aggressivita        = 0        // PER IL TEST: stesso rischio in euro per ogni trade
// 4. LIMITI
MinContratti        = 1
MaxContratti        = 80
//=========================================================================

// --- Parametri M6
UsaM6          = 1             // 0 = C1, 1 = modulo attivo
OraUscitaM6    = 170000        // ora della barra che fa uscire (C1: 210000)

// --- Parametri della strategia (C1)
EntraShort     = 1
Escludi2022    = 1             // 1 = come l'originale, per riprodurre M2
OraSegnaleS    = 040000
OraIngressoS   = 090000
OraUscitaS     = 210000

MySLS          = 3
MyTPS          = 3
SLbrkevenPS    = 2.5
TPbrkevenPS    = 2.5

// --- Posizione (dalla console)
StopS = MySLS / 100
Profitti = ProfittiPrecedenti + StrategyProfit
IF Profitti > 0 THEN
   CapitaleBase = CapitaleVersato + Aggressivita * Profitti
ELSE
   CapitaleBase = CapitaleVersato + Profitti
ENDIF
RischioEuroS = max(0, CapitaleBase) * RischioShort / 100
PsS = max(MinContratti, min(MaxContratti, RischioEuroS / (close * StopS)))

// --- M6: ora d'uscita
IF UsaM6 = 1 THEN
   OraUscitaS = OraUscitaM6
ENDIF

GiornoOk = DayOfWeek = 2

// --- Segnale delle 04:00
Time2Trade = 1
IF Escludi2022 = 1 AND (month = 4 OR month = 7) AND year = 2022 THEN
   Time2Trade = 0
ENDIF

bptns08   = low < open[1] - 0.3 * (high[1] - low[1])
range1d   = DHigh(1) - DLow(1)
body1d    = abs(DOpen(1) - DClose(1))
uptn008   = body1d > 0.5 * range1d
uptn141   = (close[1] < open[1] AND close[2] > open[2])

ONCE SegnaleMemo = 0
IF Date <> Date[1] THEN
   SegnaleMemo = 0
ENDIF
IF GiornoOk AND time = OraSegnaleS THEN
   SegnaleMemo = bptns08 AND uptn141 AND NOT uptn008
ENDIF

// --- Ingresso alle 09:00
ONCE GiornoTrade = 0
IF Time2Trade AND EntraShort AND SegnaleMemo AND GiornoOk AND time = OraIngressoS AND NOT OnMarket THEN
   SELLSHORT PsS LOTS AT MARKET
   SET STOP %LOSS MySLS
   SET TARGET %PROFIT MyTPS
   GiornoTrade = Date
ENDIF

// --- Stop e target spostati all'ingresso
IF ShortOnMarket AND SLbrkevenPS > 0 THEN
   IF (tradeprice - low) / tradeprice * 100 >= SLbrkevenPS THEN
      SET STOP BREAKEVEN
   ENDIF
ENDIF
IF ShortOnMarket AND TPbrkevenPS > 0 THEN
   IF (high - tradeprice) / tradeprice * 100 >= TPbrkevenPS THEN
      SET TARGET BREAKEVEN
   ENDIF
ENDIF

// --- Uscita alle 21:00, o al primo cambio di giorno se manca quella barra
ETimeS = (barindex - tradeindex > 17) OR time = OraUscitaS
IF ShortOnMarket AND (ETimeS OR Date <> GiornoTrade) THEN
   EXITSHORT AT MARKET
ENDIF
