//-------------------------------------------------------------------------
// 138 DAX LSD2H4 1H  |  MODULO M4  (solo long)
// Long del martedì solo dopo un lunedì in forte calo, con uscite in numeri tondi
//
// Il long dell'originale compra ogni martedì alle 04:00 e chiude alle 21:00, con target
// all'1,5%, stop all'ingresso dopo un guadagno dell'1,1% e target all'ingresso dopo una
// perdita del 2,5%. Non ha una condizione: è esposizione al DAX un giorno a settimana, con
// uscite dai valori ottimizzati. Questo modulo gli dà una ragione e toglie i numeri
// ottimizzati.
//
// Ipotesi. Dopo una giornata di forti vendite l'indice tende a recuperare il giorno
// dopo: sui futures USA il rialzo all'apertura europea (08:00-09:00) era forte proprio
// dopo le giornate di vendite (Boyarchenko, Larsen, Whelan 2023), e dal 1999 gli indici
// mondiali tendono a invertire il movimento del giorno prima, per l'arbitraggio tra
// futures ed ETF e i titoli (Baltussen, van Bekkum, Da 2019). Il "turnaround Tuesday"
// dei trader è la stessa idea applicata al lunedì. Prova debole sul DAX: l'inversione da
// un giorno all'altro è piccola in media (autocorrelazione −0,016 dopo il 1999, non
// significativa), e se c'è deve stare nei giorni di movimento grande, che è quello che
// questo modulo isola (fonti/prodotti/dax.md, ipotesi H14).
// Lunedì in forte calo: chiusura sotto l'apertura e corpo della candela oltre metà
// dell'escursione. È il pattern uptn008 dell'originale, preso con la sua direzione, ed è
// lo stesso che esclude lo short: long e short non possono scattare lo stesso martedì.
// Uscite tonde: stop al 5% come l'originale, target al 3% come lo short, niente
// spostamenti all'ingresso, chiusura alle 21:00.
// Direzione attesa: i long tenuti guadagnano in media più di quelli scartati, con molti
// meno trade, e il profitto / drawdown sale.
//
// Varianti (UsciteTonde):
//   1   condizione nuova e uscite tonde (proposta)
//   0   condizione nuova con le uscite dell'originale: separa l'effetto della condizione
//       da quello delle uscite
// Come provarlo: UsaM4 = 1 nelle due varianti e UsaM4 = 0 (riferimento: il long
// dell'originale con la taglia del test), report long. Il file è solo long
// (EntraShort = 0), quindi gli short non entrano nel confronto.
// Come giudicarlo: guadagno medio dei long scartati = (profitto riferimento − profitto
// variante 0) / (trade riferimento − trade variante 0), confrontato con quello dei
// tenuti; poi profitto / drawdown della proposta. Se la condizione non seleziona, il long
// resta esposizione al DAX e Andrea ha ragione a non usarlo.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

//=========================================================================
// CONSOLE DI GESTIONE  -  tutto in euro (Germany 40 Cash da 1 €)
//
// 1. SOLDI: CapitaleVersato = soldi messi nel sistema; ProfittiPrecedenti = guadagni
//    delle esecuzioni precedenti, da riportare a mano a ogni riavvio.
//    Backtest: CapitaleVersato = capitale del backtest, ProfittiPrecedenti = 0.
CapitaleVersato     = 10000    // PER IL TEST (capitale del backtest 10.000 €)
ProfittiPrecedenti  = 0
// 2. RISCHIO PER TRADE: % del capitale persa se scatta lo stop
RischioLong         = 15       // PER IL TEST: 1.500 € con lo stop al 5%
RischioShort        = 9        // PER IL TEST: 900 € con lo stop al 3%
// 3. MANOPOLA DEL REINVESTIMENTO: 0 = rischio fisso in euro, 1 = normale, 2 = profitti doppi
Aggressivita        = 0        // PER IL TEST: stesso rischio in euro per ogni trade
// 4. LIMITI
MinContratti        = 1
MaxContratti        = 80
//
// PER IL TEST: ogni long rischia 1.500 € allo stop del 5%, cioè una posizione da
// 30.000 € in ogni trade, come gli short delle altre proposte: 3 contratti con il DAX a
// 10.000, 1,1 a 27.000. Il minimo di 1 contratto non scatta mai.
//=========================================================================

// --- Parametri M4
UsaM4          = 1             // 0 = long dell'originale, 1 = modulo attivo
UsciteTonde    = 1             // 1 = target 3%, niente spostamenti all'ingresso; 0 = uscite originali

// --- Parametri della strategia
EntraLong      = 1             // PER IL TEST: solo long
EntraShort     = 0
Escludi2022    = 1             // 1 = come l'originale: niente trade ad aprile e luglio 2022

MySLL          = 5             // stop long, % del prezzo
MySLS          = 3             // stop short
MyTPL          = 1.5           // target long
MyTPS          = 3             // target short
SLbrkevenPL    = 1.1           // guadagno % oltre il quale lo stop del long va all'ingresso
SLbrkevenPS    = 2.5           // idem per lo short
TPbrkevenPL    = 2.5           // perdita % oltre la quale il target del long va all'ingresso
TPbrkevenPS    = 2.5           // idem per lo short

// --- M4: uscite del long
IF UsaM4 = 1 AND UsciteTonde = 1 THEN
   MyTPL       = 3
   SLbrkevenPL = 0
   TPbrkevenPL = 0
ENDIF

// --- Posizioni (dalla console)
StopL = MySLL / 100
StopS = MySLS / 100

CapitaleAttuale = CapitaleVersato + ProfittiPrecedenti + StrategyProfit
Profitti = ProfittiPrecedenti + StrategyProfit
IF Profitti > 0 THEN
   CapitaleBase = CapitaleVersato + Aggressivita * Profitti
ELSE
   CapitaleBase = CapitaleVersato + Profitti
ENDIF

RischioEuroL = max(0, CapitaleBase) * RischioLong / 100
RischioEuroS = max(0, CapitaleBase) * RischioShort / 100

PsL = max(MinContratti, min(MaxContratti, RischioEuroL / (close * StopL)))
PsS = max(MinContratti, min(MaxContratti, RischioEuroS / (close * StopS)))

// --- Segnali (come l'originale)
Time2Trade = 1
IF Escludi2022 = 1 AND (month = 4 OR month = 7) AND year = 2022 THEN
   Time2Trade = 0
ENDIF

multiplierS = 0.3
bptns08     = low < open[1] - multiplierS * (high[1] - low[1])
MyTriggerS  = bptns08

MyPtnMRL    = DayOfWeek = 2 AND time = 040000
MyPtnMRS    = DayOfWeek = 2 AND time = 040000

range1d     = DHigh(1) - DLow(1)
body1d      = abs(DOpen(1) - DClose(1))
uptn008     = body1d > 0.5 * range1d
uptn141     = (close[1] < open[1] AND close[2] > open[2])
MyPtnUNMRSY = uptn141
MyPtnUNMRSN = uptn008

// --- M4: condizione del long
LunediForteCalo = uptn008 AND DClose(1) < DOpen(1)
CondizioneL = 1
IF UsaM4 = 1 THEN
   CondizioneL = LunediForteCalo
ENDIF

// --- Ingressi
IF Time2Trade THEN
   IF EntraLong AND MyPtnMRL AND CondizioneL THEN
      BUY PsL LOTS AT MARKET
      SET STOP %LOSS MySLL
      SET TARGET %PROFIT MyTPL
   ENDIF

   IF EntraShort AND MyTriggerS AND MyPtnMRS AND MyPtnUNMRSY AND NOT MyPtnUNMRSN THEN
      SELLSHORT PsS LOTS AT MARKET
      SET STOP %LOSS MySLS
      SET TARGET %PROFIT MyTPS
   ENDIF
ENDIF

// --- Stop e target spostati all'ingresso (come l'originale; con UsciteTonde = 1 il
//     long non li usa)
IF LongOnMarket AND SLbrkevenPL > 0 THEN
   IF (high - tradeprice) / tradeprice * 100 >= SLbrkevenPL THEN
      SET STOP BREAKEVEN
   ENDIF
ENDIF
IF ShortOnMarket AND SLbrkevenPS > 0 THEN
   IF (tradeprice - low) / tradeprice * 100 >= SLbrkevenPS THEN
      SET STOP BREAKEVEN
   ENDIF
ENDIF
IF LongOnMarket AND TPbrkevenPL > 0 THEN
   IF (tradeprice - low) / tradeprice * 100 >= TPbrkevenPL THEN
      SET TARGET BREAKEVEN
   ENDIF
ENDIF
IF ShortOnMarket AND TPbrkevenPS > 0 THEN
   IF (high - tradeprice) / tradeprice * 100 >= TPbrkevenPS THEN
      SET TARGET BREAKEVEN
   ENDIF
ENDIF

// --- Uscite (come l'originale): martedì alle 21:00
ETime = (barindex - tradeindex > 17) OR time = 210000

IF LongOnMarket AND (ETime OR DayOfWeek <> 2) THEN
   SELL AT MARKET
ENDIF
IF ShortOnMarket AND (ETime OR DayOfWeek <> 2) THEN
   EXITSHORT AT MARKET
ENDIF
