//-------------------------------------------------------------------------
// 138 DAX LSD2H4 1H  |  MODULO M1
// Short senza il pattern notturno (variante: senza il filtro sul lunedì)
//
// Ipotesi. Lo short entra il martedì alle 04:00 se tra l'01:00 e le 04:00 c'è stata una
// barra in salita, poi una in discesa, poi un minimo sotto l'apertura delle 02:00 meno il
// 30% dell'escursione di quella barra (uptn141 e bptns08), e se il lunedì non è stato
// direzionale (corpo della candela non oltre metà dell'escursione, uptn008). Sono pattern
// numerati di una libreria, senza una ragione dichiarata. Quello notturno misura movimenti
// di un mercato quasi vuoto: nelle ore asiatiche l'FDAX scambia circa il 2,6% dei
// contratti della giornata (Eurex, 2024), e lo spread di IG a quell'ora è 5 punti. Inoltre
// i futures aprono all'01:10 d'inverno e alle 02:10 d'estate, quindi le stesse barre
// guardano momenti diversi secondo la stagione, e prima del dicembre 2018 di notte i
// futures erano chiusi (fonti/prodotti/dax.md, ipotesi H9 e H10).
// Direzione attesa: togliendo il pattern notturno gli short aggiunti guadagnano in media
// quanto quelli tenuti, e il sistema guadagna di più con più trade. Se invece gli short
// aggiunti perdono, il pattern seleziona davvero e resta (su 185 NQ il pattern dello short
// selezionava, M6; su 185 oro un pattern di esclusione toglieva trade perdenti, M12).
//
// Varianti (TogliNotte, TogliLunedi):
//   1, 0   senza il pattern notturno (proposta)
//   0, 1   senza il filtro sul lunedì: misura se il lunedì non direzionale seleziona
// Come provarlo: UsaM1 = 1 nelle due varianti e UsaM1 = 0 (riferimento comune a M1, M2,
// M3), report short.
// Come giudicarlo: guadagno medio degli short aggiunti = (profitto modulo − profitto
// riferimento) / (trade modulo − trade riferimento), confrontato con quello del
// riferimento; poi profitto / drawdown.
//
// La riga GRAPH mostra, il martedì alle 04:00, l'escursione in punti della barra
// 02:00-03:00 su cui si misura il pattern: serve a vedere quanto è piccola rispetto ai 5
// punti di spread. Va tolta prima del reale.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

//=========================================================================
// CONSOLE DI GESTIONE  -  tutto in euro (Germany 40 Cash da 1 €)
//
// 1. SOLDI: CapitaleVersato = soldi messi nel sistema; ProfittiPrecedenti = guadagni
//    delle esecuzioni precedenti, da riportare a mano a ogni riavvio.
//    Backtest: CapitaleVersato = capitale del backtest, ProfittiPrecedenti = 0.
CapitaleVersato     = 10000    // PER IL TEST (capitale del backtest 10.000 €)
ProfittiPrecedenti  = 0
// 2. RISCHIO PER TRADE: % del capitale persa se scatta lo stop
RischioLong         = 15       // PER IL TEST: 1.500 € con lo stop al 5%
RischioShort        = 9        // PER IL TEST: 900 € con lo stop al 3%
// 3. MANOPOLA DEL REINVESTIMENTO: 0 = rischio fisso in euro, 1 = normale, 2 = profitti doppi
Aggressivita        = 0        // PER IL TEST: stesso rischio in euro per ogni trade
// 4. LIMITI
MinContratti        = 1
MaxContratti        = 80
//
// PER IL TEST: ogni short rischia 900 € allo stop del 3%, cioè una posizione da 30.000 €
// in ogni trade: 3 contratti con il DAX a 10.000, 1,1 a 27.000. Ogni anno pesa allo
// stesso modo e il minimo di 1 contratto non scatta mai. L'originale invece entra quasi
// sempre con 1 contratto, quindi i suoi euro non sono confrontabili con questi.
//=========================================================================

// --- Parametri M1
UsaM1          = 1             // 0 = originale, 1 = modulo attivo
TogliNotte     = 1             // 1 = niente pattern notturno (uptn141 e bptns08)
TogliLunedi    = 0             // 1 = niente filtro sul lunedì (uptn008)

// --- Parametri della strategia (come l'originale)
EntraLong      = 0
EntraShort     = 1
Escludi2022    = 1             // 1 = come l'originale: niente trade ad aprile e luglio 2022

MySLL          = 5             // stop long, % del prezzo
MySLS          = 3             // stop short
MyTPL          = 1.5           // target long
MyTPS          = 3             // target short
SLbrkevenPL    = 1.1           // guadagno % oltre il quale lo stop del long va all'ingresso
SLbrkevenPS    = 2.5           // idem per lo short
TPbrkevenPL    = 2.5           // perdita % oltre la quale il target del long va all'ingresso
TPbrkevenPS    = 2.5           // idem per lo short

// --- Posizioni (dalla console)
StopL = MySLL / 100
StopS = MySLS / 100

CapitaleAttuale = CapitaleVersato + ProfittiPrecedenti + StrategyProfit
Profitti = ProfittiPrecedenti + StrategyProfit
IF Profitti > 0 THEN
   CapitaleBase = CapitaleVersato + Aggressivita * Profitti
ELSE
   CapitaleBase = CapitaleVersato + Profitti
ENDIF

RischioEuroL = max(0, CapitaleBase) * RischioLong / 100
RischioEuroS = max(0, CapitaleBase) * RischioShort / 100

PsL = max(MinContratti, min(MaxContratti, RischioEuroL / (close * StopL)))
PsS = max(MinContratti, min(MaxContratti, RischioEuroS / (close * StopS)))

// --- Segnali (come l'originale)
Time2Trade = 1
IF Escludi2022 = 1 AND (month = 4 OR month = 7) AND year = 2022 THEN
   Time2Trade = 0
ENDIF

multiplierS = 0.3
bptns08     = low < open[1] - multiplierS * (high[1] - low[1])
MyTriggerS  = bptns08

MyPtnMRL    = DayOfWeek = 2 AND time = 040000
MyPtnMRS    = DayOfWeek = 2 AND time = 040000

range1d     = DHigh(1) - DLow(1)
body1d      = abs(DOpen(1) - DClose(1))
uptn008     = body1d > 0.5 * range1d
uptn141     = (close[1] < open[1] AND close[2] > open[2])
MyPtnUNMRSY = uptn141
MyPtnUNMRSN = uptn008

// --- M1: pattern tolti
IF UsaM1 = 1 AND TogliNotte = 1 THEN
   MyTriggerS  = 1
   MyPtnUNMRSY = 1
ENDIF
IF UsaM1 = 1 AND TogliLunedi = 1 THEN
   MyPtnUNMRSN = 0
ENDIF

// --- Ingressi
IF Time2Trade THEN
   IF EntraLong AND MyPtnMRL THEN
      BUY PsL LOTS AT MARKET
      SET STOP %LOSS MySLL
      SET TARGET %PROFIT MyTPL
   ENDIF

   IF EntraShort AND MyTriggerS AND MyPtnMRS AND MyPtnUNMRSY AND NOT MyPtnUNMRSN THEN
      SELLSHORT PsS LOTS AT MARKET
      SET STOP %LOSS MySLS
      SET TARGET %PROFIT MyTPS
   ENDIF
ENDIF

// --- Stop e target spostati all'ingresso (come l'originale)
IF LongOnMarket AND SLbrkevenPL > 0 THEN
   IF (high - tradeprice) / tradeprice * 100 >= SLbrkevenPL THEN
      SET STOP BREAKEVEN
   ENDIF
ENDIF
IF ShortOnMarket AND SLbrkevenPS > 0 THEN
   IF (tradeprice - low) / tradeprice * 100 >= SLbrkevenPS THEN
      SET STOP BREAKEVEN
   ENDIF
ENDIF
IF LongOnMarket AND TPbrkevenPL > 0 THEN
   IF (tradeprice - low) / tradeprice * 100 >= TPbrkevenPL THEN
      SET TARGET BREAKEVEN
   ENDIF
ENDIF
IF ShortOnMarket AND TPbrkevenPS > 0 THEN
   IF (high - tradeprice) / tradeprice * 100 >= TPbrkevenPS THEN
      SET TARGET BREAKEVEN
   ENDIF
ENDIF

// --- Uscite (come l'originale): martedì alle 21:00
ETime = (barindex - tradeindex > 17) OR time = 210000

IF LongOnMarket AND (ETime OR DayOfWeek <> 2) THEN
   SELL AT MARKET
ENDIF
IF ShortOnMarket AND (ETime OR DayOfWeek <> 2) THEN
   EXITSHORT AT MARKET
ENDIF

// --- Solo backtest: controllo del modulo (togli prima del reale)
EscursioneNotte = 0
IF DayOfWeek = 2 AND time = 040000 THEN
   EscursioneNotte = high[1] - low[1]
ENDIF
GRAPH EscursioneNotte AS "escursione barra 02-03 del martedì (punti)"
