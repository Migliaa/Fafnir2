DEFPARAM CumulateOrders = False

EntraLong =1
EntraShort=1

TagliaPosizione=1

MySLL=0.4
MySLS=2
MyTPL=2
MyTPS=3.4

////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START INVESTMENT
SProfit             = Strategyprofit
Reinvest            = 0       //Reinvest = 1 ; No Reinvest = 0
InitCapital         = 3000    //Suggested Capital to start
RiskOnCapital       = 0.15    //% di rischio sul capitale iniziale
RiskOnProfit        = 0.2     //% di rischio sui profitti
MaxPosition         = 80      //Max contract (no more than 80 contracts)

// Gestione dei profitti separati
If Reinvest = 0 then
SProfit = 0
endif

// Calcolo del rischio totale separato
RiskCapitalBase     = InitCapital * (RiskOnCapital / 100)   // Rischio sul capitale iniziale
RiskProfit          = (SProfit / close) * (RiskOnProfit / 100)  // Rischio aggiuntivo sui profitti

// Calcolo del numero di contratti
PsL = ((RiskCapitalBase + RiskProfit) / ((close * (MySLL / 100)) * 3))
PsS = ((RiskCapitalBase + RiskProfit) / (close * (MySLS / 100)))

// Limitazioni sul numero di contratti
If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif



////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// END INVESTMENT

StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// Entrate
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
//buy TagliaPosizione contract at Market
if MySLL>0 then
set stop %loss MySLL
endif

if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
//sellshort TagliaPosizione contract at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

If longonmarket and barindex-tradeindex>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif
