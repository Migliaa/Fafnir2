//-------------------------------------------------------------------------
// 128 BIAS HOUR - AUDJPY 1H  |  VERSIONE COMBINATA C1 = originale + M4 + M6 + M8
//
// M6  reinvestimento a frazione fissa (capitale e profitto in yen)
// M8  taglia degli short inversamente proporzionale alla volatilità
// M4  niente long durante gli shock di volatilità (effetto visto su soli 3 trade)
// Ogni modulo ha il suo interruttore: con tutti a 0 il codice è l'originale con Reinvest = 0.
//
// Prima del reale:
// - togli le righe GRAPH in fondo (ProOrder non le accetta);
// - metti in CapitaleYen il valore del conto convertito in yen il giorno dell'avvio,
//   e aggiornalo se riavvii il sistema dopo un forte movimento di EURJPY;
// - controlla in un trade reale che StrategyProfit sia in yen (se fosse in euro il
//   reinvestimento resterebbe spento, senza altri effetti).
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

UsaM4 = 1
UsaM6 = 1
UsaM8 = 1

EntraLong =1
EntraShort=1

TagliaPosizione=1

MySLL=0.4
MySLS=2
MyTPL=2
MyTPS=3.4

// --- Volatilità (usata da M4 e M8)
VolShock = AverageTrueRange[24](close)      // circa 1 giorno
VolBreve = AverageTrueRange[120](close)     // circa 1 settimana
VolLunga = AverageTrueRange[2000](close)    // circa 4 mesi

// --- M4: niente long se la volatilità dell'ultimo giorno supera SogliaShockL volte quella di lungo periodo
SogliaShockL = 2
FiltroL = 1
IF UsaM4 = 1 AND VolLunga > 0 THEN
   IF VolShock / VolLunga > SogliaShockL THEN
      FiltroL = 0
   ENDIF
ENDIF

// --- M8: taglia short = taglia base * volatilità lunga / volatilità breve, tra 0.5 e 1.5
FattoreMinVol = 0.5
FattoreMaxVol = 1.5
FattoreM8 = 1
IF UsaM8 = 1 AND VolBreve > 0 AND VolLunga > 0 THEN
   FattoreM8 = max(FattoreMinVol, min(FattoreMaxVol, VolLunga / VolBreve))
ENDIF

////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START INVESTMENT
InitCapital         = 3000    //serve solo alla formula originale della taglia
RiskOnCapital       = 0.15
MaxPosition         = 80

// --- M6: taglia proporzionale al capitale, tutto in yen
CapitaleYen         = 800000  // capitale nella valuta di StrategyProfit (yen)
QuotaReinvestita    = 1
FattoreMinimo       = 0.5
FattoreMassimo      = 25

CapitaleAttuale = CapitaleYen + StrategyProfit
FattoreM6 = 1
IF UsaM6 = 1 THEN
   FattoreM6 = 1 + QuotaReinvestita * StrategyProfit / CapitaleYen
   FattoreM6 = max(FattoreMinimo, min(FattoreMassimo, FattoreM6))
ENDIF

RiskCapitalBase     = InitCapital * (RiskOnCapital / 100)

PsL = (RiskCapitalBase / ((close * (MySLL / 100)) * 3)) * FattoreM6
PsS = (RiskCapitalBase / (close * (MySLS / 100))) * FattoreM6 * FattoreM8

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif
////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// END INVESTMENT

StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// Entrate
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL AND FiltroL then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
if MySLL>0 then
set stop %loss MySLL
endif
if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

If longonmarket and barindex-tradeindex>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif

// --- Solo backtest: togli prima del reale
ONCE PiccoCapitale = CapitaleYen
PiccoCapitale = max(PiccoCapitale, CapitaleAttuale)
DDPercento = 100 * (CapitaleAttuale / PiccoCapitale - 1)
GRAPH FattoreM6 AS "fattore capitale"
GRAPH DDPercento AS "drawdown % trade chiusi"
