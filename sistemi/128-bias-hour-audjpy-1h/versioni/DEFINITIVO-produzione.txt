//-------------------------------------------------------------------------
// Codice principale : 128 BIAS HOUR - AUDJPY 1H  |  VERSIONE DI PRODUZIONE
// = originale + M4 (filtro shock sui long) + M6 (reinvestimento) + M8 (volatilità
//   sugli short) + M10 (freno sul drawdown). Versione del 2026-11-20.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

//=========================================================================
// GESTIONE DEL CAPITALE  -  tutto in YEN (la valuta in cui ProRealTime
// calcola StrategyProfit su AUDJPY)
//
// Al primo avvio:      CapitaleYen = conto convertito in yen, gli altri due a 0.
// Se fermi e riavvii:  StrategyProfit riparte da zero, quindi riporta qui a mano
//   ProfittoPrecedente = somma dei profitti realizzati (in yen) nelle esecuzioni
//                        precedenti, dal report ProOrder
//   PiccoPrecedente    = capitale massimo raggiunto finora, in yen
//                        (CapitaleYen + profitto realizzato nel momento migliore);
//                        0 se non vuoi riportarlo (il freno ripartirà dal capitale attuale)
// Se versi o prelevi soldi, aggiorna CapitaleYen e lascia ProfittoPrecedente com'è.
//=========================================================================
CapitaleYen         = 800000
ProfittoPrecedente  = 0
PiccoPrecedente     = 0

QuotaReinvestita    = 1       // 1 = taglia proporzionale al capitale; 0.5 = reinvesti metà dei profitti; 0 = taglia fissa
MaxDD               = 0.40    // drawdown dal massimo a cui la taglia arriva al minimo del freno
FrenoMinimo         = 0.4     // taglia minima in drawdown, come frazione di quella piena
//=========================================================================

EntraLong  = 1
EntraShort = 1

MySLL = 0.4
MySLS = 2
MyTPL = 2
MyTPS = 3.4

// --- Capitale, reinvestimento (M6) e freno sul drawdown (M10)
ProfittoTotale  = ProfittoPrecedente + StrategyProfit
CapitaleAttuale = CapitaleYen + ProfittoTotale

FattoreCapitale = 1 + QuotaReinvestita * ProfittoTotale / CapitaleYen
FattoreCapitale = max(0.5, min(25, FattoreCapitale))

ONCE PiccoCapitale = max(PiccoPrecedente, CapitaleYen + ProfittoPrecedente)
PiccoCapitale = max(PiccoCapitale, CapitaleAttuale)

Freno = (CapitaleAttuale - (1 - MaxDD) * PiccoCapitale) / (MaxDD * PiccoCapitale)
Freno = max(FrenoMinimo, min(1, Freno))

// --- Volatilità (M4 e M8)
VolShock = AverageTrueRange[24](close)      // circa 1 giorno
VolBreve = AverageTrueRange[120](close)     // circa 1 settimana
VolLunga = AverageTrueRange[2000](close)    // circa 4 mesi

// M4: niente long se la volatilità dell'ultimo giorno è più del doppio di quella di lungo periodo
FiltroL = 1
IF VolLunga > 0 THEN
   IF VolShock / VolLunga > 2 THEN
      FiltroL = 0
   ENDIF
ENDIF

// M8: taglia short moltiplicata per volatilità lunga / volatilità breve, tra 0.5 e 1.5
FattoreVolS = 1
IF VolBreve > 0 AND VolLunga > 0 THEN
   FattoreVolS = max(0.5, min(1.5, VolLunga / VolBreve))
ENDIF

// --- Taglie (formula originale x fattori)
InitCapital     = 3000
RiskOnCapital   = 0.15
MaxPosition     = 80

RiskCapitalBase = InitCapital * (RiskOnCapital / 100)

PsL = (RiskCapitalBase / ((close * (MySLL / 100)) * 3)) * FattoreCapitale * Freno
PsS = (RiskCapitalBase / (close * (MySLS / 100))) * FattoreCapitale * Freno * FattoreVolS

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif

// --- Segnali (invariati rispetto all'originale)
StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// --- Entrate
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL AND FiltroL then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
if MySLL>0 then
set stop %loss MySLL
endif
if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

// --- Uscite
If longonmarket and barindex-tradeindex>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif
