//-------------------------------------------------------------------------
// 128 BIAS HOUR - AUDJPY 1H  |  MODULO M11 (SPERIMENTALE)
// Ciclo settimanale del rischio: il long riassorbe la riduzione di fine settimana,
// lo short anticipa quella successiva
//
// Ipotesi. Prima del weekend il mercato resta chiuso circa 48 ore e chi tiene posizioni
// di carry su AUDJPY non può coprirsi: una parte riduce l'esposizione tra giovedì e
// venerdì e la ricostruisce dalla riapertura. È un flusso prevedibile che si ripete
// ogni settimana. Nel modello di Brunnermeier-Pedersen (2005) chi conosce in anticipo
// una vendita forzata vende prima e ricompra dopo, il prezzo va oltre il valore
// giusto e poi rientra. Le due gambe del sistema sono le due metà di questo ciclo:
// lo short del giovedì vende prima della riduzione, il long del lunedì compra il rientro.
// Il pattern trovato dal generatore va nella stessa direzione: uptn049 esclude il long
// dopo tre chiusure giornaliere in rialzo, cioè quando prima del weekend non c'è stata
// nessuna vendita da riassorbire. M11 misura direttamente il movimento:
//   long  (lunedì 03:00): solo se il prezzo è sceso rispetto al giovedì 12:00 precedente
//   short (giovedì 12:00): solo se il prezzo è salito rispetto al lunedì 03:00
// La soglia è zero, quindi non c'è nulla da ottimizzare.
// Se l'ipotesi è giusta, il guadagno medio dei trade tenuti sale e quello dei trade
// scartati scende. Se è sbagliata succede come con M5: i due medi restano simili.
// Riferimenti: Brunnermeier-Pedersen 2005; Dao-McGroarty-Urquhart 2016 (i salti del
// weekend sulle valute tendono a rientrare nella settimana). La riduzione delle posizioni
// prima del weekend è una pratica dei desk, non un fatto dimostrato da studi.
//
// Gli interruttori ApplicaLong e ApplicaShort permettono di provare i due lati separati.
// UsaM11 = 0 riproduce esattamente l'originale con Reinvest = 0.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False

UsaM11 = 1                // 0 = originale, 1 = modulo attivo

EntraLong =1
EntraShort=1

TagliaPosizione=1

MySLL=0.4
MySLS=2
MyTPL=2
MyTPS=3.4

// --- Parametri M11
ApplicaLong        = 1    // 1 = condizione sul long
ApplicaShort       = 1    // 1 = condizione sullo short
QuotaSenzaConferma = 0    // taglia dei trade senza conferma: 0 = saltati, 0.5 = metà taglia

// --- Prezzi alle due ore di ingresso, registrati ogni settimana anche senza trade
ONCE PrezzoLunedi  = 0
ONCE PrezzoGiovedi = 0
ONCE MovDaLunedi   = 0
ONCE MovDaGiovedi  = 0

IF DayOfWeek = 4 AND Time = 120000 THEN
   IF PrezzoLunedi > 0 THEN
      MovDaLunedi = close - PrezzoLunedi       // lunedì 03:00 -> giovedì 12:00
   ENDIF
   PrezzoGiovedi = close
ENDIF

IF DayOfWeek = 1 AND Time = 030000 THEN
   IF PrezzoGiovedi > 0 THEN
      MovDaGiovedi = close - PrezzoGiovedi     // giovedì 12:00 -> lunedì 03:00, weekend compreso
   ENDIF
   PrezzoLunedi = close
ENDIF

FattoreL = 1
FattoreS = 1
IF UsaM11 = 1 THEN
   IF ApplicaLong = 1 AND MovDaGiovedi >= 0 THEN    // nessun calo da riassorbire
      FattoreL = QuotaSenzaConferma
   ENDIF
   IF ApplicaShort = 1 AND MovDaLunedi <= 0 THEN    // nessun rialzo da scaricare
      FattoreS = QuotaSenzaConferma
   ENDIF
ENDIF

////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START INVESTMENT
InitCapital         = 3000
RiskOnCapital       = 0.15
MaxPosition         = 80

RiskCapitalBase     = InitCapital * (RiskOnCapital / 100)

PsL = (RiskCapitalBase / ((close * (MySLL / 100)) * 3)) * FattoreL
PsS = (RiskCapitalBase / (close * (MySLS / 100))) * FattoreS

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif
////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// END INVESTMENT

StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// Entrate (FattoreL / FattoreS = 0 significa trade saltato)
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL AND FattoreL > 0 then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
if MySLL>0 then
set stop %loss MySLL
endif
if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS AND FattoreS > 0 then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

If longonmarket and barindex-tradeindex>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif
