//-------------------------------------------------------------------------
// 128 BIAS HOUR - AUDJPY 1H  |  MODULO M8
// Taglia a volatilità obiettivo applicata solo agli short
//
// Ipotesi ricavata dal fallimento di M2. Il long ha uno stop stretto (0,4%): la sua
// perdita massima è fissata dallo stop e la tua formula la tiene già costante, quindi
// scalare la taglia con la volatilità introduce solo variazioni di rischio senza
// informazione. Lo short invece ha uno stop di emergenza (2%) che quasi mai scatta
// ed esce a tempo dopo 29 barre: la sua perdita reale dipende da quanto si muove il
// prezzo in quelle 29 ore, cioè dalla volatilità. È il caso in cui la gestione per
// volatilità ha senso.
// Oggi la taglia dello short è calcolata sullo stop del 2%, che non rappresenta il
// rischio vero; qui viene moltiplicata per volatilità di lungo periodo / volatilità
// recente, tra FattoreMinVol e FattoreMaxVol. I long sono identici all'originale.
// UsaM8 = 0 riproduce esattamente l'originale con Reinvest = 0.
// Fonti: Harvey et al. 2018 (su valute: meno code e drawdown, rendimento simile).
// Da giudicare sul report delle sole posizioni short.
//
// Le righe GRAPH in fondo servono solo nel backtest: toglile prima del reale.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

UsaM8 = 1                 // 0 = originale, 1 = modulo attivo

EntraLong =1
EntraShort=1

TagliaPosizione=1

MySLL=0.4
MySLS=2
MyTPL=2
MyTPS=3.4

// --- Parametri M8
BarreVolBreve  = 120      // circa 1 settimana di barre orarie
BarreVolLunga  = 2000     // circa 4 mesi di barre orarie
FattoreMinVol  = 0.5
FattoreMaxVol  = 1.5

VolBreve = AverageTrueRange[BarreVolBreve](close)
VolLunga = AverageTrueRange[BarreVolLunga](close)

FattoreM8 = 1
IF UsaM8 = 1 AND VolBreve > 0 AND VolLunga > 0 THEN
   FattoreM8 = max(FattoreMinVol, min(FattoreMaxVol, VolLunga / VolBreve))
ENDIF

////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START INVESTMENT
InitCapital         = 3000
RiskOnCapital       = 0.15
MaxPosition         = 80

RiskCapitalBase     = InitCapital * (RiskOnCapital / 100)

PsL = RiskCapitalBase / ((close * (MySLL / 100)) * 3)
PsS = (RiskCapitalBase / (close * (MySLS / 100))) * FattoreM8

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif
////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// END INVESTMENT

StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// Entrate
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
if MySLL>0 then
set stop %loss MySLL
endif
if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

If longonmarket and barindex-tradeindex>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif

// --- Solo backtest: controllo del modulo (togli prima del reale)
GRAPH FattoreM8 AS "fattore taglia short"
