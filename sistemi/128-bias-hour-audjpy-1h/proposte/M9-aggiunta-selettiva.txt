// 128 BIAS HOUR - AUDJPY 1H  |  MODULO M9
// M7 rivisto: aggiunta solo se il long conferma in fretta, e di metà taglia
//
// Perché M7 non funzionava: con una volatilità oraria di AUDJPY intorno allo 0,14%,
// anche un prezzo che si muove a caso arriva a +0,4% (una volta lo stop) entro 40 ore
// circa due volte su tre. M7 ha aggiunto in circa metà dei long: la condizione non
// selezionava le settimane buone, aggiungeva esposizione quasi a caso, e infatti ha
// alzato profitto e drawdown insieme, peggiorando il loro rapporto.
// Entro 8 ore il caso porta a +1 volta lo stop circa una volta su tre: la conferma
// rapida è più informativa. L'aggiunta è di metà taglia e lo stop della posizione è
// messo dove la perdita totale resta quella di un long normale.
// Gli short non sono toccati.
// UsaM9 = 0 riproduce esattamente l'originale con Reinvest = 0.
//
// COME LEGGERE IL REPORT: ProRealTime conta l'aggiunta come trade separato, quindi il
// "guadagno medio per trade" scende anche se ogni posizione guadagna di più. Il numero
// di aggiunte è (trade di M9 - trade dell'originale); il guadagno medio per posizione è
// profitto totale / trade dell'originale. Il confronto giusto per il drawdown è con
// l'originale ingrandito dello stesso profitto in più: se M9 guadagna il 10% in più,
// il suo drawdown deve restare sotto quello originale * 1,10.
//
// Le righe GRAPH in fondo servono solo nel backtest: toglile prima del reale.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = True   // necessario per poter aggiungere al long

UsaM9 = 1                 // 0 = originale, 1 = modulo attivo

EntraLong =1
EntraShort=1

TagliaPosizione=1

MySLL=0.4
MySLS=2
MyTPL=2
MyTPS=3.4

// --- Parametri M9 (prova BarreMaxAggiunta 8 / 4 e SogliaR 1 / 1.5, non altri valori)
SogliaR          = 1      // aggiungi quando il long guadagna SogliaR volte la distanza dello stop
BarreMaxAggiunta = 8      // ... entro queste barre dall'entrata
QuotaAggiunta    = 0.5    // taglia dell'aggiunta rispetto alla prima entrata
RischioDopoR     = 1      // perdita massima della posizione dopo l'aggiunta, in perdite a stop originali

////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START INVESTMENT
InitCapital         = 3000
RiskOnCapital       = 0.15
MaxPosition         = 80

RiskCapitalBase     = InitCapital * (RiskOnCapital / 100)

PsL = RiskCapitalBase / ((close * (MySLL / 100)) * 3)
PsS = RiskCapitalBase / (close * (MySLS / 100))

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif
////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// END INVESTMENT

// --- M9: stato della posizione long
// (TRADEINDEX dopo l'aggiunta indicherebbe la barra dell'aggiunta: per l'uscita
// a 66 barre si usa la barra della prima entrata, salvata qui)
IF LongOnMarket AND NOT LongOnMarket[1] THEN
   PrezzoIngresso = PositionPrice
   BarraIngresso  = BarIndex
   TagliaIniziale = CountOfLongShares
   Aggiunto       = 0
ENDIF
IF NOT LongOnMarket THEN
   Aggiunto = 0
ENDIF
IF LongOnMarket AND CountOfLongShares > TagliaIniziale THEN
   Aggiunto = 1
ENDIF

StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// Entrate
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL AND NOT LongOnMarket then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
if MySLL>0 then
set stop %loss MySLL
endif
if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS AND NOT ShortOnMarket then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

// --- M9: aggiunta (confermata a chiusura di barra, eseguita in apertura della successiva)
IF UsaM9 = 1 AND LongOnMarket AND Aggiunto = 0 AND (BarIndex - BarraIngresso) <= BarreMaxAggiunta THEN
   LivelloAggiunta = PrezzoIngresso * (1 + SogliaR * MySLL / 100)
   IF close >= LivelloAggiunta THEN
      PsAgg = max(1, TagliaIniziale * QuotaAggiunta)
      buy PsAgg Lots at Market
   ENDIF
ENDIF

// --- M9: stop dell'intera posizione dopo l'aggiunta
// prezzo medio - distanza stop * RischioDopoR * taglia iniziale / taglia totale
IF UsaM9 = 1 AND LongOnMarket AND Aggiunto = 1 THEN
   DistStopIniziale  = PrezzoIngresso * MySLL / 100
   LivelloProtezione = PositionPrice - DistStopIniziale * RischioDopoR * TagliaIniziale / CountOfLongShares
   IF close > LivelloProtezione THEN
      sell at LivelloProtezione stop
   ELSE
      sell at market
   ENDIF
ENDIF

If longonmarket and barindex-BarraIngresso>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif

// --- Solo backtest: togli prima del reale
GRAPH CountOfLongShares AS "contratti long"
GRAPH Aggiunto AS "aggiunta fatta"
