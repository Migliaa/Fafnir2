//-------------------------------------------------------------------------
// 128 BIAS HOUR - AUDJPY 1H  |  MODULO M2
// Taglia inversamente proporzionale alla volatilità recente (volatility targeting)
//
// Cosa cambia rispetto all'originale: SOLO la taglia. Stop e target restano in %.
// Quando la volatilità dell'ultima settimana è il doppio di quella degli ultimi
// mesi la taglia si dimezza, quando è la metà la taglia sale (fino a FattoreMaxVol).
// UsaM2 = 0 riproduce esattamente l'originale con Reinvest = 0.
// Fonti: Moreira-Muir 2017, Harvey et al. 2018.
// ATTENZIONE: M2 e M3 agiscono sulla stessa leva (rischio per trade). Non combinarli
// direttamente: tieni quello dei due che regge meglio.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

UsaM2 = 1                 // 0 = originale, 1 = modulo attivo

EntraLong =1
EntraShort=1

TagliaPosizione=1

MySLL=0.4
MySLS=2
MyTPL=2
MyTPS=3.4

// --- Parametri M2
BarreVolBreve  = 120      // circa 1 settimana di barre orarie
BarreVolLunga  = 2000     // circa 4 mesi di barre orarie
FattoreMinVol  = 0.5      // taglia minima rispetto all'originale
FattoreMaxVol  = 1.5      // taglia massima rispetto all'originale (metti 1 per non superare mai l'originale)

VolBreve = AverageTrueRange[BarreVolBreve](close)
VolLunga = AverageTrueRange[BarreVolLunga](close)

RapportoVol = 1
IF VolLunga > 0 THEN
   RapportoVol = VolBreve / VolLunga
ENDIF

FattoreM2 = 1
IF UsaM2 = 1 AND RapportoVol > 0 THEN
   FattoreM2 = max(FattoreMinVol, min(FattoreMaxVol, 1 / RapportoVol))
ENDIF

////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// START INVESTMENT
InitCapital         = 3000
RiskOnCapital       = 0.15
MaxPosition         = 80

RiskCapitalBase     = InitCapital * (RiskOnCapital / 100)

PsL = (RiskCapitalBase / ((close * (MySLL / 100)) * 3)) * FattoreM2
PsS = (RiskCapitalBase / (close * (MySLS / 100))) * FattoreM2

If PsL > MaxPosition then
PsL = MaxPosition
endif

If PsS > MaxPosition then
PsS = MaxPosition
endif

If PsL < 1 then
PsL = 1
endif

If PsS < 1 then
PsS = 1
endif
////==//==//==//==//==//==//==//==//==//==//==//==//==//==//==// END INVESTMENT

StartTrade = 010000
EndTrade   = 220000

TriggerL=DayOfWeek=1 AND Time=030000
TriggerS=DayOfWeek=4 AND Time=120000

ExitL =LongOnMarket  AND DayOfWeek=3 AND Time=220000
ExitS =ShortOnMarket AND DayOfWeek=5 AND Time=180000

uptn091 = high[0]<high[1]
uptn049 = Dclose(1)>Dclose(2) and Dclose(2)>Dclose(3) and Dclose(3)>Dclose(4)
if TriggerL then
PtnMRLY   = 1
PtnMRLN   = uptn049
PtnUNMRLY = 1
PtnUNMRLN = uptn091
endif

uptn134 = (Dhigh(2)>Dhigh(1) and Dlow(2)<Dlow(1))
uptn142 = (close[1] > open[1] and  close[2] < open[2])
if TriggerS then
PtnMRSY   = 1
PtnMRSN   = uptn134
PtnUNMRSY = 1
PtnUNMRSN = uptn142
endif

// Entrate
if time >= StartTrade and time <= EndTrade then
if EntraLong AND TriggerL then
If (PtnMRLY AND not PtnMRLN) AND (PtnUNMRLY AND not PtnUNMRLN) then
buy PsL Lots at Market
if MySLL>0 then
set stop %loss MySLL
endif
if MyTPL>0 then
set target %profit MyTPL
endif
endif
endif

if EntraShort AND TriggerS then
If (PtnMRSY AND not PtnMRSN) AND (PtnUNMRSY AND not PtnUNMRSN) then
sellshort PsS Lots at Market
if MySLS>0 then
set stop %loss MySLS
endif
if MyTPS>0 then
set target %profit MyTPS
endif
endif
endif
endif

If longonmarket and barindex-tradeindex>66 then
sell at market
endif

If shortonmarket and barindex-tradeindex>29 then
exitshort at market
endif

If ExitL then
sell at market
endif

If ExitS then
exitshort at market
endif
