//-------------------------------------------------------------------------
// 097 BRK IN GBPJPY 30m  |  MODULO M4 (sopra C0)
// Cosa porta il pattern che esclude le aperture di Londra calme
//
// Il pattern uptn133 dell'originale salta la giornata quando la prima mezz'ora di
// Londra (09:00-09:30) ha un'escursione inferiore al 75% della mezz'ora precedente
// (08:30-09:00) e la mezz'ora successiva (09:30-10:00) non si allarga di più del 10%.
// Viene da una libreria di pattern numerati, senza una ragione dichiarata: può essere
// informazione o una scelta fatta sul backtest (Bailey et al.; Neely-Weller 2003 sulle
// regole intraday del forex cercate tra molte combinazioni). Il contributo dei pattern
// dei sistemi di Andrea non è mai stato misurato (P1-P3 di idee/idee-di-trading.md), e
// le esclusioni fisse vanno riverificate (principio 4).
// Ipotesi a favore del pattern. Volumi e volatilità del forex hanno un picco
// all'apertura di Londra (Ito-Hashimoto 2006): se la prima mezz'ora è più calma di
// quella prima, Londra non sta partecipando, e una rottura senza Londra dietro ha meno
// acquisti che la portano avanti (fonti/prodotti/gbpjpy.md, A2 e A3).
// Direzione attesa: i trade che il pattern scarta guadagnano meno di quelli tenuti.
// Se guadagnano uguale, il pattern non contiene informazione e toglierlo aggiunge trade.
//
// Nessuna variante: UsaM4 = 1 toglie il pattern, UsaM4 = 0 riproduce C0.
// Come provarlo: UsaM4 = 1 e riferimento UsaM4 = 0, report long.
// Come giudicarlo: guadagno medio dei trade aggiunti =
//   (profitto M4 - profitto riferimento) / (trade M4 - trade riferimento),
// contro il guadagno medio del riferimento; i trade aggiunti saranno pochi, quindi
// conta anche quanti sono (lezione M4 di 128 BIAS HOUR).
//
// Le righe GRAPH in fondo servono solo nel backtest: toglile prima del reale.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False

UsaM4 = 1                      // 0 = C0, 1 = pattern uptn133 tolto

//=========================================================================
// GESTIONE DEL CAPITALE  -  tutto in yen, la valuta di StrategyProfit su GBP/JPY
//
// Primo avvio:  Capitale = soldi del conto destinati al sistema, convertiti in yen;
//               ProfittoPrecedente e PiccoPrecedente a 0.
// Riavvio:      StrategyProfit riparte da zero, quindi riporta a mano
//               ProfittoPrecedente = profitti realizzati nelle esecuzioni precedenti
//               PiccoPrecedente    = capitale più alto raggiunto finora (serve solo
//                                    con il freno; 0 = riparte dal capitale attuale)
// Versamenti o prelievi: cambia Capitale, lascia ProfittoPrecedente com'è.
// Backtest:     Capitale = capitale del backtest, gli altri due a 0.
//=========================================================================
Capitale            = 1000000  // PER IL TEST, in yen
ProfittoPrecedente  = 0
PiccoPrecedente     = 0

Reinvesti           = 0        // PER IL TEST; 1 = posizione proporzionale al capitale attuale
Leva                = 2        // valore della posizione / capitale, a peso 1
ContrattiFissi      = 0        // con Reinvesti = 0: contratti a peso 1; 0 = valore fisso Leva × Capitale

PesoLong            = 1        // moltiplicatore della posizione long
PesoShort           = 1        // non usato: il sistema è solo long

UsaFreno            = 0        // 1 = posizione ridotta durante i drawdown (Grossman-Zhou)
MaxDD               = 0.40     // drawdown dal massimo a cui la posizione arriva a FrenoMinimo
FrenoMinimo         = 0.4

MinContratti        = 0.5      // verificare il minimo di IG sul mini
MaxContratti        = 80
ValorePerPunto      = 10000    // yen per 1,00 di prezzo, 1 contratto mini (10.000 GBP)
//
// Con lo stop allo 0,75% il rischio di uno stop pieno è Leva × 0,75% del capitale:
// Leva 2 = 1,5%, quanto l'originale con 1 contratto e 6.000 euro.
//=========================================================================

// --- Parametri della strategia (come C0)
EntraLong          = 1
IngressoEsplicito  = 0         // 1 = se alle 10:30 il prezzo è già sopra il livello, compra al mercato
DistanzaMinima     = 0         // con IngressoEsplicito = 1: compra al mercato anche se il prezzo è
                               // sotto il livello di meno di questo (in yen di prezzo: 0.05 = 5 pip)

MySLL       = 0.75             // stop, % del prezzo d'ingresso
MyTPL       = 2                // target
OraLivello  = 033000           // fine della notte asiatica: massimo della giornata fin qui
OraIngresso = 103000           // ordine stop alla chiusura di questa barra, valido mezz'ora
OraUscitaL  = 220000
BarreMaxL   = 22

// --- Capitale attuale, freno e posizioni
CapitaleAttuale = Capitale + ProfittoPrecedente + StrategyProfit

ONCE PiccoCapitale = max(PiccoPrecedente, Capitale + ProfittoPrecedente)
PiccoCapitale = max(PiccoCapitale, CapitaleAttuale)

Freno = 1
IF UsaFreno = 1 THEN
   Freno = (CapitaleAttuale - (1 - MaxDD) * PiccoCapitale) / (MaxDD * PiccoCapitale)
   Freno = max(FrenoMinimo, min(1, Freno))
ENDIF

IF Reinvesti = 1 THEN
   ContrattiBase = Leva * CapitaleAttuale / (close * ValorePerPunto)
ELSE
   IF ContrattiFissi > 0 THEN
      ContrattiBase = ContrattiFissi
   ELSE
      ContrattiBase = Leva * Capitale / (close * ValorePerPunto)
   ENDIF
ENDIF

PsL = max(MinContratti, min(MaxContratti, ContrattiBase * PesoLong * Freno))

// --- Massimo della notte asiatica
IF Time = OraLivello THEN
   LivelloHigh = DHigh(0)
ENDIF

// --- Segnale (come l'originale)
TriggerL = Time = OraIngresso
GiorniL  = DayOfWeek = 1 OR DayOfWeek = 3

IF Day = 24 AND Month = 6 AND Year = 2016 THEN
   TriggerL = 0
ENDIF

// niente long se l'apertura di Londra (09:00-09:30) è stata più calma della mezz'ora
// prima e la mezz'ora dopo non si è allargata
uptn133 = (High[1] - Low[1] < (High[2] - Low[2]) * 1.1) AND ((High[2] - Low[2]) < (High[3] - Low[3]) * 0.75)

IF UsaM4 = 1 THEN
   uptn133 = 0
ENDIF

// --- Ingresso
IF EntraLong AND TriggerL AND GiorniL AND NOT uptn133 THEN
   IF IngressoEsplicito = 1 AND close >= LivelloHigh - DistanzaMinima THEN
      BUY PsL LOTS AT MARKET
   ELSE
      BUY PsL LOTS AT LivelloHigh STOP
   ENDIF
   SET STOP %LOSS MySLL
   SET TARGET %PROFIT MyTPL
ENDIF

// --- Uscita
IF LongOnMarket AND (Time = OraUscitaL OR BarIndex - TradeIndex > BarreMaxL) THEN
   SELL AT MARKET
ENDIF

// --- Solo backtest (togli prima del reale)
GRAPH uptn133 AS "pattern attivo"
