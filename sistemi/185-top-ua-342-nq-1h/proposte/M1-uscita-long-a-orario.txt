//-------------------------------------------------------------------------
// 185 TOP UA 342 NQ 1H  |  MODULO M1
// Uscita del long a un giorno e un orario fissi invece che dopo 90 barre
//
// Il long entra il lunedì ed esce dopo 90 barre orarie, cioè il venerdì mattina più o
// meno all'ora d'ingresso, oppure prima a stop o target. Le varianti dicono in quale
// tratto della settimana sta il guadagno (principio 1 di idee/idee-di-trading.md).
//
// Varianti (GiorniDopoL, OraUscitaL):
//   1, 220000   martedì alla chiusura americana (proposta). Ipotesi nata dai report di
//               Andrea, non da una fonte indipendente: il long del lunedì guadagna 55 € a
//               trade, quello del martedì 24 € pur restando aperto quasi altrettanto, fino
//               al venerdì. Se i due trade condividono il tratto da martedì a venerdì, la
//               differenza sta nelle prime 24 ore circa. Uscire il martedì sera tiene quel
//               tratto, toglie tre notti di finanziamento (circa 0,05% del valore della
//               posizione) e tre giorni di rischio, e libera la posizione per una taglia
//               più grande. Direzione attesa: guadagno medio più basso ma molto meno che
//               in proporzione al tempo in posizione, drawdown molto più basso, profitto /
//               drawdown più alto.
//   1, 090000   martedì alle 09:00: lunedì più una notte, fino all'apertura europea, dove
//               dal 1998 al 2020 i futures USA hanno fatto il rendimento più alto della
//               giornata (Boyarchenko, Larsen, Whelan 2023; fonti/prodotti/nasdaq.md, A1).
//   4, 090000   venerdì alle 09:00: come l'originale, ma sempre dopo l'apertura europea
//               invece che a un'ora che dipende dall'ingresso.
// Con le prime due varianti il long non è aperto quando di solito scatta lo short, quindi
// cambiano anche gli short che oggi invertono un long: va letto anche il report short.
// Come provarlo: UsaM1 = 1 e UsaM1 = 0 (riferimento comune a tutta la tornata), report
// long e short.
// Come giudicarlo: stessi ingressi, quindi guadagno medio, drawdown, profitto / drawdown
// e tempo in posizione. La differenza di profitto tra due varianti è quanto rende il
// tratto di posizione che le separa.
//
// Le righe GRAPH in fondo servono solo nel backtest: toglile prima del reale.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False
DEFPARAM PreLoadBars = 3000

//=========================================================================
// GESTIONE DEL CAPITALE  -  tutto in euro (US Tech 100 Cash da 1 €, valuta di StrategyProfit)
//
// Primo avvio:  Capitale = euro del conto destinati al sistema, ProfittoPrecedente
//               e PiccoPrecedente a 0.
// Riavvio:      StrategyProfit riparte da zero, quindi riporta a mano
//               ProfittoPrecedente = profitti realizzati nelle esecuzioni precedenti
//               PiccoPrecedente    = capitale più alto raggiunto finora (serve solo
//                                    con il freno; 0 = riparte dal capitale attuale)
// Versamenti o prelievi: cambia Capitale, lascia ProfittoPrecedente com'è.
// Backtest:     Capitale = capitale del backtest, gli altri due a 0.
//=========================================================================
Capitale            = 5000
ProfittoPrecedente  = 0
PiccoPrecedente     = 0

Reinvesti           = 0        // 1 = posizione proporzionale al capitale attuale; 0 = non cresce con i profitti
Leva                = 6        // PER IL TEST: valore della posizione / capitale, a peso 1
ContrattiFissi      = 0        // con Reinvesti = 0: contratti a peso 1; 0 = valore fisso Leva × Capitale

PesoLong            = 1        // moltiplicatore della posizione long
PesoShort           = 1        // PER IL TEST (originale: 2, cioè rischio dell'8% invece del 4%)

UsaFreno            = 0        // 1 = posizione ridotta durante i drawdown (Grossman-Zhou)
MaxDD               = 0.40     // drawdown dal massimo a cui la posizione arriva a FrenoMinimo
FrenoMinimo         = 0.4

MinContratti        = 0.5
MaxContratti        = 80
ValorePerPunto      = 1        // valore di 1 punto per 1 contratto (US Tech 100 Cash 1 € -> 1)
//
// PER IL TEST: 6 × 5.000 = posizione da 30.000 in ogni trade, circa 1 contratto ai
// prezzi di oggi e 7 nel 2016, così ogni anno pesa allo stesso modo e il minimo di 0,5
// contratti non scatta mai; un trade che arriva allo stop del 4% perde 1.200. Con
// Reinvesti = 0 il capitale del backtest non entra nei calcoli: se ProRealTime ferma il
// backtest per capitale insufficiente, alzalo.
//=========================================================================

// --- Parametri M1
UsaM1          = 1             // 0 = originale, 1 = modulo attivo
GiorniDopoL    = 1             // giorni di calendario dopo quello d'ingresso (0 = stesso giorno)
OraUscitaL     = 220000        // ora della barra che fa scattare l'uscita
BarreSicurezzaL = 200          // con M1 attivo sostituisce le 90 barre (il venerdì alle 21:00 chiude comunque)

// --- Parametri della strategia (come l'originale)
EntraLong     = 1
EntraShort    = 1

MySLL         = 4          // stop long, % del prezzo
MySLS         = 4          // stop short
MyTPL         = 2.5        // target long
MyTPS         = 1.6        // target short

MyStartTradeL = 080000
MyEndTradeL   = 200000
MyStartTradeS = 160000
MyEndTradeS   = 200000
PortATRMeno   = 4          // calo minimo dall'apertura della giornata, in escursioni medie orarie
BarreUscitaL  = 90
BarreUscitaS  = 100
MaxDayInTradeL = 8          // giorni in posizione (contati come nell'originale)
MaxDayInTradeS = 4
ONCE DayInTrade = 0

// --- Capitale attuale, freno e posizioni (intestazione standard)
CapitaleAttuale = Capitale + ProfittoPrecedente + StrategyProfit

ONCE PiccoCapitale = max(PiccoPrecedente, Capitale + ProfittoPrecedente)
PiccoCapitale = max(PiccoCapitale, CapitaleAttuale)

Freno = 1
IF UsaFreno = 1 THEN
   Freno = (CapitaleAttuale - (1 - MaxDD) * PiccoCapitale) / (MaxDD * PiccoCapitale)
   Freno = max(FrenoMinimo, min(1, Freno))
ENDIF

IF Reinvesti = 1 THEN
   ContrattiBase = Leva * CapitaleAttuale / (close * ValorePerPunto)
ELSE
   IF ContrattiFissi > 0 THEN
      ContrattiBase = ContrattiFissi
   ELSE
      ContrattiBase = Leva * Capitale / (close * ValorePerPunto)
   ENDIF
ENDIF

PsL = max(MinContratti, min(MaxContratti, ContrattiBase * PesoLong * Freno))
PsS = max(MinContratti, min(MaxContratti, ContrattiBase * PesoShort * Freno))

// --- Segnali (come l'originale)
GiornoLong = dayofweek = 1

shortAvg  = Average[5](close)
shortAvg2 = Average[10](close)
iptn18    = shortAvg crosses over shortAvg2 and close > (shortAvg * 0.99)

uptn116    = (dhigh(1) - dclose(1)) < 0.20 * (dhigh(1) - dlow(1))
MyTriggerS = close < DOpen(0) - PortATRMeno * AverageTrueRange[20](close)

// --- Ingressi (come l'originale)
IF time >= MyStartTradeL AND time <= MyEndTradeL THEN
   IF EntraLong AND iptn18 AND GiornoLong THEN
      BUY PsL LOTS AT MARKET
      SET STOP %LOSS MySLL
      SET TARGET %PROFIT MyTPL
   ENDIF
ENDIF

IF time >= MyStartTradeS AND time <= MyEndTradeS THEN
   IF EntraShort AND MyTriggerS AND uptn116 THEN
      SELLSHORT PsS LOTS AT MARKET
      SET STOP %LOSS MySLS
      SET TARGET %PROFIT MyTPS
   ENDIF
ENDIF

// --- M1: giorni di calendario passati dall'ingresso del long
ONCE GiorniL = 0
IF LongOnMarket THEN
   IF NOT LongOnMarket[1] THEN
      GiorniL = 0
   ELSE
      IF Date <> Date[1] THEN
         GiorniL = GiorniL + 1
      ENDIF
   ENDIF
ENDIF

UscitaM1 = 0
BarreL = BarreUscitaL
IF UsaM1 = 1 THEN
   BarreL = BarreSicurezzaL
   IF LongOnMarket THEN
      IF GiorniL > GiorniDopoL OR (GiorniL = GiorniDopoL AND time >= OraUscitaL) THEN
         UscitaM1 = 1
      ENDIF
   ENDIF
ENDIF

// --- Uscite (come l'originale, più M1)
// giorni in posizione, come l'originale: il contatore riparte solo quando si passa da
// fuori a dentro il mercato, quindi uno short che inverte un long eredita i giorni del long
IF OnMarket <> OnMarket[1] THEN
   DayInTrade = 0
ENDIF
IF OnMarket AND (Day > Day[1] OR (DayInTrade > 3 AND DayOfWeek < 1)) THEN
   DayInTrade = DayInTrade + 1
ENDIF
IF LongOnMarket AND DayInTrade >= MaxDayInTradeL AND MaxDayInTradeL > 0 THEN
   SELL AT MARKET
ENDIF
IF ShortOnMarket AND DayInTrade >= MaxDayInTradeS AND MaxDayInTradeS > 0 THEN
   EXITSHORT AT MARKET
ENDIF

FineSettimana = dayofweek = 5 AND time = 210000

IF LongOnMarket AND (barindex - tradeindex > BarreL OR FineSettimana OR UscitaM1) THEN
   SELL AT MARKET
ENDIF

IF ShortOnMarket AND (barindex - tradeindex > BarreUscitaS OR FineSettimana) THEN
   EXITSHORT AT MARKET
ENDIF

// --- Solo backtest: controllo del modulo (togli prima del reale)
GRAPH GiorniL AS "giorni dall'ingresso del long"
GRAPH UscitaM1 AS "uscita M1"
