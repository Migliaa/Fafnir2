// Uscite a tempo spostate attorno alla sessione asiatica
//
// Ipotesi. L'oro sale in media nelle ore asiatiche (23:00-09:00 ora italiana) e scende
// in quelle di Londra: lo trovano Blose, Gondhalekar e Kort 2018 sui futures COMEX e sul
// fixing di Londra, una tesi della Copenhagen Business School sul 2001-2018 (salita fino
// alle 08:30 circa) e i dati del World Gold Council sul primo semestre 2026 (Asia +12,9%,
// Nord America -15%). Il meccanismo è la domanda fisica di Cina e India, che compra nelle
// sue ore (fonti/prodotti/oro.md, A1).
// - Long: l'originale esce alla prima barra del quarto giorno, verso l'01:00, cioè
//   all'inizio della salita asiatica e subito dopo la pausa serale, quando lo spread è
//   più largo. Qui esce lo stesso giorno alle OraUscitaL (09:00, apertura di Londra).
// - Short: l'originale esce dopo 60 barre, di solito tra la notte e la mattina del terzo
//   giorno, dopo tre notti asiatiche. Qui esce alle OraUscitaS (21:00) del secondo
//   giorno dopo l'ingresso (NottiS = 2), prima della terza notte.
// Stop, target e le altre uscite dell'originale restano; i long in perdita dopo 80
// barre escono come prima, spesso prima delle 09:00.
// Direzione attesa: guadagno medio un po' più alto su entrambi i lati. L'effetto atteso
// è piccolo, circa lo 0,05% del prezzo per notte asiatica (2 $ a contratto con l'oro a
// 4.000 $), quindi conta che non peggiori e che regga anno per anno.
// UsaM1 = 0 riproduce l'originale a 1 contratto (BuyLot = 0).
//
// Varianti: OraUscitaL 090000 (proposta), 080000. OraUscitaS 210000 (proposta), 160000
// (dopo l'asta del pomeriggio di Londra, il minimo della giornata nella tesi CBS).
// NottiS 2 (proposta), 1.
// Come provarlo: lanciare UsaM1 = 1 e UsaM1 = 0 (riferimento), report long e short
// separati. Il riferimento è lo stesso di M2 e M3.
// Come giudicarlo, per ciascun lato: i trade sono quasi gli stessi, quindi conta la
// differenza di guadagno medio e di profitto a parità di numero di trade, poi le barre
// annuali: il miglioramento deve esserci anche negli anni dal 2019.
//
// Le righe GRAPH in fondo servono solo nel backtest: toglile prima del reale.
//-------------------------------------------------------------------------
DEFPARAM CumulateOrders = False

//=========================================================================
// GESTIONE DEL CAPITALE  -  tutto in dollari (valuta dell'oro e di StrategyProfit)
//
// Primo avvio:  Capitale = dollari del conto destinati al sistema (gli euro convertiti
//               al cambio del giorno), ProfittoPrecedente e PiccoPrecedente a 0.
// Riavvio:      StrategyProfit riparte da zero, quindi riporta a mano
//               ProfittoPrecedente = profitti realizzati nelle esecuzioni precedenti
//               PiccoPrecedente    = capitale più alto raggiunto finora (serve solo
//                                    con il freno; 0 = riparte dal capitale attuale)
// Versamenti o prelievi: cambia Capitale, lascia ProfittoPrecedente com'è.
// Backtest:     Capitale = capitale del backtest, gli altri due a 0.
//=========================================================================
Capitale            = 4000     // PER IL TEST: capitale del backtest, in dollari
ProfittoPrecedente  = 0        // PER IL TEST (originale: 200)
PiccoPrecedente     = 0

Reinvesti           = 0        // PER IL TEST: contratti fissi; 1 = posizione proporzionale al capitale attuale
Leva                = 4        // con Reinvesti = 1: valore della posizione / capitale, a peso 1
ContrattiFissi      = 1        // con Reinvesti = 0: contratti a peso 1

PesoLong            = 1        // moltiplicatore della posizione long
PesoShort           = 1        // moltiplicatore della posizione short

UsaFreno            = 0        // 1 = posizione ridotta durante i drawdown
MaxDD               = 0.40     // drawdown dal massimo a cui la posizione arriva a FrenoMinimo
FrenoMinimo         = 0.4

MinContratti        = 0.5
MaxContratti        = 80
ValorePerPunto      = 1        // dollari per 1 $ di movimento di 1 contratto (Spot Gold $1 di IG)
//
// Leva 4 vale circa il 4% di rischio dell'originale con uno stop all'1%: quale delle due
// formule tenere lo decide M4.
//=========================================================================

// --- Parametri M1
UsaM1          = 1             // 0 = originale, 1 = modulo attivo
ApplicaLongM1  = 1
ApplicaShortM1 = 1
OraUscitaL     = 090000        // long: ora d'uscita del giorno di scadenza
OraUscitaS     = 210000        // short: ora d'uscita...
NottiS         = 2             // ...del giorno in cui il contatore dei giorni arriva a NottiS

// --- Parametri della strategia (come l'originale)
EntraLong       = 1
EntraShort      = 1
Escludi2011     = 1            // 1 = come l'originale (niente trade da giugno a novembre 2011)

SLmultL   = 3
SLmultS   = 3
TPmultL   = 13
TPmultS   = 0
ATRperiod = 20

MyStartTrade     = 110000
MyEndTrade       = 220000
MaxDayInTradeL   = 4
MaxDayInTradeS   = 4
PortATRpiu       = 2.5
PortATRMeno      = 5.5
Once DayInTrade  = 0

// --- Stop e target in percentuale, legati all'escursione media delle ultime 20 ore
ATRpts = AverageTrueRange[ATRperiod](close)
MySLL = (SLmultL * ATRpts) / close * 100
MySLS = (SLmultS * ATRpts) / close * 100
MyTPL = (TPmultL * ATRpts) / close * 100
MyTPS = (TPmultS * ATRpts) / close * 100

// --- Capitale attuale, freno e posizioni (intestazione standard)
CapitaleAttuale = Capitale + ProfittoPrecedente + StrategyProfit

ONCE PiccoCapitale = max(PiccoPrecedente, Capitale + ProfittoPrecedente)
PiccoCapitale = max(PiccoCapitale, CapitaleAttuale)

Freno = 1
IF UsaFreno = 1 THEN
   Freno = (CapitaleAttuale - (1 - MaxDD) * PiccoCapitale) / (MaxDD * PiccoCapitale)
   Freno = max(FrenoMinimo, min(1, Freno))
ENDIF

IF Reinvesti = 1 THEN
   ContrattiBase = Leva * CapitaleAttuale / (close * ValorePerPunto)
ELSE
   ContrattiBase = ContrattiFissi
ENDIF

PsL = max(MinContratti, min(MaxContratti, ContrattiBase * PesoLong * Freno))
PsS = max(MinContratti, min(MaxContratti, ContrattiBase * PesoShort * Freno))

// --- Ingressi (come l'originale)
Time2Trade = 1
IF Escludi2011 = 1 AND Year = 2011 AND Month >= 6 AND Month <= 11 THEN
   Time2Trade = 0
ENDIF

IF Time2Trade THEN

MyTriggerL = close > DOpen(0) + PortATRpiu  * AverageTrueRange[200](close)
MyTriggerS = close < DOpen(0) - PortATRMeno * AverageTrueRange[200](close)

ptnl25 = high[0] > (high[1] + high[1] * 0.5 * 0.01)
ptns25 = low[0] < (low[1] - low[1] * 0.5 * 0.01)

uptn104 = Dlow(0) > (Dlow(1) + Dlow(1) * 0.01)
uptn139 = (Dclose(1) < Dopen(1) AND Dclose(2) > Dopen(2))

IF time >= MyStartTrade AND time <= MyEndTrade THEN

// --- LONG
IF EntraLong AND MyTriggerL AND ptnl25 AND NOT uptn104 THEN
BUY PsL LOTS AT MARKET
if MySLL > 0 then
set stop %loss MySLL
endif
if MyTPL > 0 then
set target %profit MyTPL
endif
ENDIF

// --- SHORT
IF EntraShort AND MyTriggerS AND ptns25 AND NOT uptn139 THEN
SELLSHORT PsS LOTS AT MARKET
if MySLS > 0 then
set stop %loss MySLS
endif
if MyTPS > 0 then
set target %profit MyTPS
endif
ENDIF
ENDIF
ENDIF

// --- Uscite
IF OnMarket <> OnMarket[1] THEN
DayInTrade = 0
ENDIF

IF OnMarket AND Day > Day[1] THEN
DayInTrade = DayInTrade + 1
ENDIF

// Long: scadenza a tempo (M1: alle OraUscitaL invece che alla prima barra del giorno)
ScadenzaL = DayInTrade >= MaxDayInTradeL
IF UsaM1 = 1 AND ApplicaLongM1 = 1 THEN
   ScadenzaL = DayInTrade >= MaxDayInTradeL AND time >= OraUscitaL
ENDIF
IF LongOnMarket AND ScadenzaL THEN
SELL AT MARKET
ENDIF

IF ShortOnMarket AND DayInTrade >= MaxDayInTradeS THEN
EXITSHORT AT MARKET
ENDIF

// Short (M1): uscita alle OraUscitaS del giorno NottiS dopo l'ingresso
IF UsaM1 = 1 AND ApplicaShortM1 = 1 AND ShortOnMarket AND DayInTrade >= NottiS AND time >= OraUscitaS THEN
EXITSHORT AT MARKET
ENDIF

IF LongOnMarket THEN
EL0 = barindex - tradeindex > 106 OR (barindex - tradeindex > 80 AND PositionPerf(0) < 0)
IF EL0 THEN
SELL AT MARKET
ENDIF
ENDIF

IF ShortOnMarket THEN
ES0 = barindex - tradeindex > 60
IF ES0 THEN
EXITSHORT AT MARKET
ENDIF
ENDIF

// --- Solo backtest: controllo (togli prima del reale)
GRAPH DayInTrade AS "giorni in posizione"
GRAPH MySLL AS "stop long %"
