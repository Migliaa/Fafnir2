// 3251_ORAF_FDAX_5M_TUTTI1400-1635-S · Intraday a orario + filtri e uscite · FDAX (DAX Futures (Eurex)) · short
// Codice interno: FDAX-ORF-C1D605
// Grafico: CFD DAX Futures (Eurex), 5 minuti, ora italiana. Gli orari dei parametri sono di Francoforte:
//   uguali all'ora italiana, ora legale compresa. Sessione RTH: 0900-1730 di Francoforte.
// Taglia: RischioTrade = 0% del capitale a ogni stop (0 = sempre ContrattiFissi), freno a 50% di drawdown;
//   commissioni, spread e interessi si impostano in ProRealTime.
// Le uscite di fine sessione o di fine settimana sono anticipate all'apertura dell'ultima barra.
// Generato da AlphaMining il 2026-11-11 dal giro 8. Si rigenera, non si ritocca.
//-------------------------------------------------------------------------------------------------
// Sezioni: 1 parametri di base, 2 parametri del sistema, 3 money management, 4 stop e target,
//          5 taglia, 6 filtri e livelli, 7 uscite, 8 ingressi (nell'ordine del motore).
//-------------------------------------------------------------------------------------------------
DEFPARAM CumulateOrders = False

//==//==//==//==//==//==//==// 1. PARAMETRI DI BASE
EntraLong = 0           // 1 = apre posizioni long, 0 = no
EntraShort = 1          // 1 = apre posizioni short, 0 = no

//==//==//==//==//==//==//==// 2. PARAMETRI DEL SISTEMA (orari di Francoforte: uguali all'ora italiana)
Giorni = 2 //inutile              // 0 = tutti i giorni
OraIngresso = 1400      // a mercato, all'apertura della prima barra da qui (Francoforte)
OraUscita = 1635        // a mercato, all'apertura della prima barra da qui (Francoforte)

//==//==//==//==//==//==//==// 3. MONEY MANAGEMENT
Reinvest        = 1           // 1 = la taglia cresce col capitale; 0 = sempre ContrattiFissi
ContrattiFissi  = 1           // contratti per ordine quando Reinvest = 0
Capitale        = 10000       // capitale iniziale: lo stesso impostato nel backtest PRT
RischioTrade    = 2.5+ (strategyProfit/2000)           // % del capitale persa quando il trade chiude al suo stop (0 = da scegliere)
DrawdownMax     = 30          // freno: oltre questa % sotto il massimo non apre piu' nulla (0 = spento)
Passo           = 0.5         // la taglia cresce a passi di ...
TagliaMin       = 0.5         // taglia minima
MaxContratti    = 80          // tetto ai contratti per ordine, da impostare a mano
TettoContratti  = 80          // limite del broker: MaxContratti non puo' superarlo
ValorePunto     = 1           // EUR per 1,00 di prezzo con 1 contratto
// Esempio: Capitale 5000 e RischioTrade 3 -> ogni stop costa circa 150 EUR. Un gap oltre lo stop costa di piu'.
// Cosa comporta RischioTrade: la tabella (crescita e drawdown per ogni valore) arriva con la
// scelta sull'IS della lista dei trade a 1 contratto.

//==//==//==//==//==//==//==// 4. STOP E TARGET (in punti o in % dell'ingresso)
StopPunti = 167      // stop loss in punti (0 = no)
StopPerc = 0       // stop loss in % dell'ingresso (0 = no)
TargetPunti = 0    // take profit in punti (0 = no)
TargetPerc = 0     // take profit in % (0 = no)
TrailingPerc = 0  // trailing stop in % dell'ingresso, dal miglior prezzo (0 = no)
PareggioPerc = 0.31  // stop a pareggio dopo un guadagno in % (0 = no)

// ordini di stop e target per la barra dopo, ricalcolati a ogni chiusura
IF OnMarket THEN
IF LongOnMarket THEN
lato = 1
ELSE
lato = -1
ENDIF
ingresso = PositionPrice
IF BarIndex = TradeIndex(1) THEN
migliore = ingresso                  // il miglior prezzo parte dall'ingresso
ELSIF lato = 1 THEN
migliore = max(migliore, High)
ELSE
migliore = min(migliore, Low)
ENDIF
mioStop = 0
IF (StopPunti > 0) OR (StopPerc > 0) THEN
mioStop = ingresso - lato * (StopPunti + StopPerc / 100 * ingresso)
ENDIF
IF TrailingPerc > 0 THEN
trail = migliore - lato * TrailingPerc / 100 * ingresso
IF (mioStop = 0) OR (lato * (trail - mioStop) > 0) THEN
mioStop = trail
ENDIF
ENDIF
IF (PareggioPerc > 0) AND (lato * (migliore - ingresso) >= PareggioPerc / 100 * ingresso) AND ((mioStop = 0) OR (lato * (ingresso - mioStop) > 0)) THEN
mioStop = ingresso
ENDIF
mioTarget = 0
IF (TargetPunti > 0) OR (TargetPerc > 0) THEN
mioTarget = ingresso + lato * (TargetPunti + TargetPerc / 100 * ingresso)
ENDIF
IF lato = 1 THEN
IF mioStop > 0 THEN
SELL AT mioStop STOP
ENDIF
IF mioTarget > 0 THEN
SELL AT mioTarget LIMIT
ENDIF
ELSE
IF mioStop > 0 THEN
EXITSHORT AT mioStop STOP
ENDIF
IF mioTarget > 0 THEN
EXITSHORT AT mioTarget LIMIT
ENDIF
ENDIF
ENDIF

distL = StopPunti + StopPerc / 100 * Close     // distanza dello stop in punti, per la taglia
distS = distL

//==//==//==//==//==//==//==// 5. TAGLIA (ricalcolata a ogni barra con l'utile gia' chiuso)
capitaleOra = Capitale + STRATEGYPROFIT
// freno sul drawdown: si misura dal massimo del capitale; una volta scattato resta fermo
// (si riattiva solo riavviando il sistema o alzando DrawdownMax)
ONCE Picco = Capitale
ONCE Fermo = 0
IF capitaleOra > Picco THEN
Picco = capitaleOra
ENDIF
DDOra = (Picco - capitaleOra) / Picco * 100
IF DrawdownMax > 0 AND DDOra >= DrawdownMax THEN
Fermo = 1
ENDIF
// contratti perche' uno stop costi RischioTrade % del capitale: distanza dello stop in euro per 1 contratto
IF Reinvest = 0 OR RischioTrade <= 0 THEN
TagliaL = ContrattiFissi
TagliaS = ContrattiFissi
ELSE
TagliaL = floor(capitaleOra * RischioTrade / 100 / (distL * ValorePunto) / Passo) * Passo
TagliaS = floor(capitaleOra * RischioTrade / 100 / (distS * ValorePunto) / Passo) * Passo
IF TagliaL < TagliaMin THEN
TagliaL = TagliaMin
ENDIF
IF TagliaS < TagliaMin THEN
TagliaS = TagliaMin
ENDIF
ENDIF
Tetto = min(MaxContratti, TettoContratti)
IF TagliaL > Tetto THEN
TagliaL = Tetto
ENDIF
IF TagliaS > Tetto THEN
TagliaS = Tetto
ENDIF

//==//==//==//==//==//==//==// 6. FILTRI E LIVELLI
// ==== ORA DI FRANCOFORTE: il motore ragiona li', il grafico e' in ora italiana
// Francoforte e l'Italia hanno lo stesso fuso e la stessa ora legale: nessuna conversione
differenza = 0
minChi = Hour * 60 + Minute - differenza     // chiusura della barra, minuti dalla mezzanotte di Francoforte
giornoChi = DayOfWeek                        // giorno della settimana a Francoforte (0 = domenica)
IF minChi < 0 THEN
minChi = minChi + 1440
giornoChi = giornoChi - 1
IF giornoChi < 0 THEN
giornoChi = 6
ENDIF
ENDIF
oraChi = floor(minChi / 60) * 100 + (minChi MOD 60)
fineSessione = (minChi = 1050)             // ultima barra della sessione RTH
inSessione = (((minChi - 5 - 540 + 1440) MOD 1440) < 510)    // barre della sessione RTH
// sul CFD ci sono barre anche fuori dalla sessione: la sessione nuova comincia dalla prima barra dentro, e la barra
// che chiude all'apertura decide gli ingressi della prima barra (2026-11-06: il 2220 si fermava dopo 3 trade)
ONCE fuoriSessione = 0
IF NOT inSessione THEN
fuoriSessione = 1
ENDIF
// una pausa che contiene l'apertura (es. festivi USA: chiusura alle 12 e riapertura alle 17) apre una sessione nuova
// anche se la barra prima non era l'ultima per orario (lezione 2026-11-09: sessioni festive fuse con la dopo)
pausa = ((minChi - 5 - minChi[1] + 1440) MOD 1440)
nuovaSessione = inSessione AND (fineSessione[1] OR (NOT inSessione[1]) OR ((pausa > 0) AND (((540 - minChi[1] + 1440) MOD 1440) <= pausa)))
preApertura = ((NOT inSessione) AND (minChi = 540))
// barre della sessione contate: le uscite «dopo N barre» contano solo queste, come il motore (lezione 2026-11-09,
// 2515 FDAX: contando anche le ore di notte del CFD si usciva dopo 24 ore invece che dopo 24 barre del future)
ONCE barreSess = 0
IF inSessione THEN
barreSess = barreSess + 1
ENDIF
IF OnMarket AND (BarIndex = TradeIndex(1)) THEN
barraInS = barreSess
ENDIF

// ==== CANDELE E FILTRI (pattern della libreria, stesse regole del motore)
IF inSessione THEN
// ---- candele giornaliere: GO0, GH0, GL0 la candela in corso; $GO[k] ... $GC[k] le chiuse
ONCE GNuova = 1
ONCE GN = 0
ONCE GBarre = 0
ONCE GRotta = -1000
ONCE GIntera = 1
GSessRotta = nuovaSessione AND (GBarre < 81.6)      // la sessione appena finita era a meta'
IF nuovaSessione THEN
GBarre = 0
ENDIF
GBarre = GBarre + 1
IF GSessRotta THEN
IF GNuova THEN                       // era nella candela chiusa alla barra prima
GRotta = GN
ELSE
GIntera = 0
ENDIF
ENDIF
GSalto = (nuovaSessione) AND NOT GNuova        // la candela prima non si e' chiusa: si chiude ora
IF GSalto THEN
$GO[1] = GO0
$GH[1] = GH0
$GL[1] = GL0
$GC[1] = Close[1]
GN = GN + 1
IF NOT GIntera THEN
GRotta = GN
ENDIF
GNuova = 1
ENDIF
IF GNuova THEN
GO0 = Open
GH0 = High
GL0 = Low
GIntera = 1
ELSE
GH0 = max(GH0, High)
GL0 = min(GL0, Low)
ENDIF
GFine = (minChi = 1050)
IF GFine THEN                          // la candela e' chiusa: diventa la numero 1
$GO[1] = GO0
$GH[1] = GH0
$GL[1] = GL0
$GC[1] = Close
GN = GN + 1
IF (NOT GIntera) OR (GBarre < 81.6) THEN
GRotta = GN
ENDIF
ENDIF
GNuova = GFine

// Filtro 1: P06 no (giornaliero)
filtro1S = ((GN >= 1) AND (GN - GRotta >= 1) AND NOT GFine AND NOT (Close < $GL[1]))
// Filtro 2: P60 sì (giornaliero)
filtro2S = NOT GFine AND (GH0 - Close > 0.005 * GH0)
ENDIF

// ==== SOLO LE BARRE DELLA SESSIONE RTH (0900-1730 di Francoforte)
IF inSessione OR preApertura THEN
IF preApertura THEN                          // la barra dopo apre la sessione di oggi (CFD con barre fuori sessione)
prossimaOra = 0900
prossimoGiorno = giornoChi
ELSIF fineSessione AND fuoriSessione THEN    // la barra dopo e' fuori sessione: si decide alla barra prima dell'apertura
prossimaOra = -1
prossimoGiorno = -1
ELSIF fineSessione THEN                      // la barra dopo apre la sessione di domani (lunedi' dopo il venerdi')
prossimaOra = 0900
IF giornoChi = 5 THEN
prossimoGiorno = 1
ELSE
prossimoGiorno = giornoChi + 1
ENDIF
ELSE
prossimaOra = oraChi
prossimoGiorno = giornoChi
ENDIF
giornoOk = ((prossimoGiorno = 2)OR (prossimoGiorno = 4) OR (prossimoGiorno = 5))
//==//==//==//==//==//==//==// 7. USCITE
// ==== USCITA: all'apertura della prima barra dall'orario di uscita; al piu' tardi all'apertura dell'ultima
IF OnMarket AND ((prossimaOra >= OraUscita) OR (minChi = 1045)) THEN
IF LongOnMarket THEN
SELL AT MARKET
ELSE
EXITSHORT AT MARKET
ENDIF
ENDIF

//==//==//==//==//==//==//==// 8. INGRESSI
// ==== INGRESSO: una volta al giorno, all'apertura della prima barra dall'orario d'ingresso
ONCE entrato = 0
IF fineSessione THEN
entrato = 0                              // domani e' un altro giorno
ENDIF
IF (entrato = 0) AND (prossimaOra >= OraIngresso) AND (prossimaOra < OraUscita) THEN
entrato = 1                              // se il filtro dice no, quel giorno non si entra piu'
IF giornoOk AND filtro1S AND filtro2S THEN
IF NOT Fermo AND EntraShort THEN
SELLSHORT TagliaS CONTRACTS AT MARKET
ENDIF
ENDIF
ENDIF
ENDIF
